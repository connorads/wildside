const SAMPLE_RATE: u64 = 44_100;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Scenario {
    BootTitleIdle10s,
    BootTitleStartLevel1,
    BootTitleStartLevel1MoveRight5s,
    Level1EnemyWindow,
}

#[derive(Debug, Clone, Copy)]
struct FrameCheckpoint {
    label: &'static str,
    frame_index: usize,
}

#[derive(Debug, Serialize)]
struct OracleManifest {
    sample_rate: u32,
    audio_path: PathBuf,
    frames: Vec<OracleFrame>,
    states: Vec<OracleState>,
}

#[derive(Debug, Serialize)]
struct OracleFrame {
    label: String,
    frame_index: usize,
    png_path: PathBuf,
    sha1: String,
}

#[derive(Debug, Clone, Copy, Serialize, PartialEq, Eq, PartialOrd, Ord)]
#[serde(rename_all = "snake_case")]
enum EnemyFacing {
    Left,
    Right,
}

#[derive(Debug, Serialize)]
struct OracleEnemyState {
    object_type: u8,
    x_pos: u32,
    y_pos: u32,
    x_velocity: i32,
    y_velocity: i32,
    facing: EnemyFacing,
}

#[derive(Debug, Serialize)]
struct OracleState {
    label: String,
    frame_index: usize,
    game_mode: u16,
    current_player: u8,
    number_hitpoints: u16,
    current_helmet_id: u16,
    current_level_id: u16,
    player_1_level_id: u16,
    camera_x_pos: u32,
    camera_y_pos: u32,
    kid_x_pos: u32,
    kid_y_pos: u32,
    number_objects: u16,
    number_gfx_objects: u16,
    active_enemies: Vec<OracleEnemyState>,
}

const GAME_MODE_ADDR: usize = 0xFF_FBCA;
const CURRENT_PLAYER_ADDR: usize = 0xFF_FC39;
const NUMBER_HITPOINTS_ADDR: usize = 0xFF_FC40;
const CURRENT_LEVEL_ID_ADDR: usize = 0xFF_FC44;
const CURRENT_HELMET_ID_ADDR: usize = 0xFF_FC46;
const PLAYER_1_LEVEL_ID_ADDR: usize = 0xFF_FC5A;
const CAMERA_X_POS_ADDR: usize = 0xFF_F81C;
const CAMERA_Y_POS_ADDR: usize = 0xFF_F820;
const KID_X_POS_ADDR: usize = 0xFF_24FA;
const KID_Y_POS_ADDR: usize = 0xFF_24FE;
const NUMBER_OBJECTS_ADDR: usize = 0xFF_F84A;
const NUMBER_GFX_OBJECTS_ADDR: usize = 0xFF_F85C;
const GFX_OBJECT_RAM_ADDR: usize = 0xFFFF_2448;
const GFX_OBJECT_SLOT_SIZE: usize = 0x4C;
const GFX_OBJECT_SLOT_COUNT: usize = 0x4D;
const DISPLAY_FLAG_OFFSET: usize = 0x13;
const X_DIRECTION_OFFSET: usize = 0x16;
const OBJECT_X_POS_OFFSET: usize = 0x1A;
const OBJECT_Y_POS_OFFSET: usize = 0x1E;
const OBJECT_X_VEL_OFFSET: usize = 0x26;
const OBJECT_Y_VEL_OFFSET: usize = 0x2A;
const OBJECT_META_OFFSET: usize = 0x3A;

struct NullSaveWriter;

impl SaveWriter for NullSaveWriter {
    type Err = io::Error;

    fn load_bytes(&mut self, _: &str) -> Result<Vec<u8>, Self::Err> {
        Err(io::Error::other("save loading is disabled"))
    }

    fn persist_bytes(&mut self, _: &str, _: &[u8]) -> Result<(), Self::Err> {
        Ok(())
    }

    fn load_serialized<D: Decode<()>>(&mut self, _: &str) -> Result<D, Self::Err> {
        Err(io::Error::other("save loading is disabled"))
    }

    fn persist_serialized<E: Encode>(&mut self, _: &str, _: E) -> Result<(), Self::Err> {
        Ok(())
    }
}

#[derive(Default)]
struct CaptureRenderer {
    width: u32,
    height: u32,
    rgb: Vec<u8>,
}

impl Renderer for CaptureRenderer {
    type Err = io::Error;

    fn render_frame(
        &mut self,
        frame_buffer: &[Color],
        frame_size: FrameSize,
        _: RenderFrameOptions,
    ) -> Result<(), Self::Err> {
        self.width = frame_size.width;
        self.height = frame_size.height;
        self.rgb.clear();
        self.rgb.reserve_exact(frame_size.width as usize * frame_size.height as usize * 3);
        for color in frame_buffer
            .iter()
            .take(frame_size.width as usize * frame_size.height as usize)
        {
            self.rgb.extend_from_slice(&[color.r, color.g, color.b]);
        }
        Ok(())
    }
}

#[derive(Default)]
struct AudioCollector {
    samples: Vec<f32>,
}

impl AudioOutput for AudioCollector {
    type Err = io::Error;

    fn push_sample(&mut self, sample_l: f64, sample_r: f64) -> Result<(), Self::Err> {
        self.samples.push(sample_l as f32);
        self.samples.push(sample_r as f32);
        Ok(())
    }
}

fn main() -> Result<()> {
    let mut args = env::args().skip(1);
    let mut rom_path = None;
    let mut out_dir = None;
    let mut scenario = None;

    while let Some(arg) = args.next() {
        match arg.as_str() {
            "--rom" => rom_path = args.next().map(PathBuf::from),
            "--out-dir" => out_dir = args.next().map(PathBuf::from),
            "--scenario" => scenario = args.next().map(|value| parse_scenario(&value)),
            other => return Err(anyhow!("unknown arg {other}")),
        }
    }

    let rom_path = rom_path.context("missing --rom")?;
    let out_dir = out_dir.context("missing --out-dir")?;
    let scenario = scenario.transpose()?.context("missing --scenario")?;
    fs::create_dir_all(&out_dir)
        .with_context(|| format!("failed to create {}", out_dir.display()))?;

    let rom = fs::read(&rom_path)
        .with_context(|| format!("failed to read ROM from {}", rom_path.display()))?;
    let script = scenario_script(scenario);
    let checkpoints = scenario_checkpoints(scenario);
    let mut checkpoint_index = 0usize;

    let mut save_writer = NullSaveWriter;
    let mut emulator = GenesisEmulator::create(rom, genesis_config(), &mut save_writer);
    emulator.update_audio_output_frequency(SAMPLE_RATE);
    let mut renderer = CaptureRenderer::default();
    let mut audio = AudioCollector::default();
    let mut frame_index = 0usize;
    let mut rendered_frames = Vec::new();
    let mut captured_states = Vec::new();

    while frame_index < scenario_frame_count(scenario) {
        let inputs = inputs_from_mask(script.get(frame_index).copied().unwrap_or(0));
        loop {
            if emulator.tick(&mut renderer, &mut audio, &inputs, &mut save_writer)? ==
                TickEffect::FrameRendered
            {
                while checkpoint_index < checkpoints.len()
                    && checkpoints[checkpoint_index].frame_index == frame_index
                {
                    let checkpoint = checkpoints[checkpoint_index];
                    let png_path = out_dir.join(format!(
                        "{:04}_{}.png",
                        checkpoint.frame_index, checkpoint.label
                    ));
                    write_png(&png_path, renderer.width, renderer.height, &renderer.rgb)?;
                    rendered_frames.push(OracleFrame {
                        label: checkpoint.label.to_string(),
                        frame_index: checkpoint.frame_index,
                        sha1: sha1_hex(&renderer.rgb),
                        png_path,
                    });
                    captured_states.push(capture_state(
                        &mut emulator,
                        checkpoint.label,
                        checkpoint.frame_index,
                    ));
                    checkpoint_index += 1;
                }
                frame_index += 1;
                break;
            }
        }
    }

    let audio_path = out_dir.join("audio.wav");
    write_wav(&audio_path, &audio.samples)?;
    let manifest = OracleManifest {
        sample_rate: SAMPLE_RATE as u32,
        audio_path,
        frames: rendered_frames,
        states: captured_states,
    };
    println!("{}", serde_json::to_string_pretty(&manifest)?);
    Ok(())
}

fn parse_scenario(value: &str) -> Result<Scenario> {
    match value {
        "boot_title_idle_10s" => Ok(Scenario::BootTitleIdle10s),
        "boot_title_start_level1" => Ok(Scenario::BootTitleStartLevel1),
        "boot_title_start_level1_move_right_5s" => Ok(Scenario::BootTitleStartLevel1MoveRight5s),
        "level1_enemy_window" => Ok(Scenario::Level1EnemyWindow),
        _ => Err(anyhow!("unknown scenario {value}")),
    }
}

fn scenario_frame_count(scenario: Scenario) -> usize {
    match scenario {
        Scenario::BootTitleIdle10s => 600,
        Scenario::BootTitleStartLevel1 => 720,
        Scenario::BootTitleStartLevel1MoveRight5s => 1_020,
        Scenario::Level1EnemyWindow => 900,
    }
}

fn scenario_checkpoints(scenario: Scenario) -> &'static [FrameCheckpoint] {
    match scenario {
        Scenario::BootTitleIdle10s => &[
            FrameCheckpoint {
                label: "title",
                frame_index: 360,
            },
            FrameCheckpoint {
                label: "title_tail",
                frame_index: 599,
            },
        ],
        Scenario::BootTitleStartLevel1 => &[
            FrameCheckpoint {
                label: "title",
                frame_index: 360,
            },
            FrameCheckpoint {
                label: "level_entry",
                frame_index: 520,
            },
            FrameCheckpoint {
                label: "level_settle",
                frame_index: 719,
            },
        ],
        Scenario::BootTitleStartLevel1MoveRight5s => &[
            FrameCheckpoint {
                label: "title",
                frame_index: 360,
            },
            FrameCheckpoint {
                label: "level_entry",
                frame_index: 520,
            },
            FrameCheckpoint {
                label: "move_mid",
                frame_index: 840,
            },
            FrameCheckpoint {
                label: "move_end",
                frame_index: 1_019,
            },
        ],
        Scenario::Level1EnemyWindow => &[
            FrameCheckpoint {
                label: "level_settle",
                frame_index: 719,
            },
            FrameCheckpoint {
                label: "enemy_window_open",
                frame_index: 810,
            },
            FrameCheckpoint {
                label: "enemy_window_tail",
                frame_index: 899,
            },
        ],
    }
}

fn scenario_script(scenario: Scenario) -> Vec<u8> {
    let mut script = vec![0; scenario_frame_count(scenario).saturating_add(1)];
    match scenario {
        Scenario::BootTitleIdle10s => {}
        Scenario::BootTitleStartLevel1 => {
            apply_single_frames(&mut script, &[120, 200, 280, 360, 440], 0x80);
            apply_range(&mut script, 460, 460, 0x80);
        }
        Scenario::BootTitleStartLevel1MoveRight5s => {
            apply_single_frames(&mut script, &[120, 200, 280, 360, 440], 0x80);
            apply_range(&mut script, 460, 460, 0x80);
            apply_range(&mut script, 720, 1_019, 1 << 3);
        }
        Scenario::Level1EnemyWindow => {
            apply_single_frames(&mut script, &[120, 200, 280, 360, 440], 0x80);
            apply_range(&mut script, 460, 460, 0x80);
            apply_range(&mut script, 720, 899, 1 << 3);
        }
    }
    script
}

fn capture_state(
    emulator: &mut GenesisEmulator,
    label: &str,
    frame_index: usize,
) -> OracleState {
    let debug = emulator.debug();
    let memory = debug.memory_view(GenesisMemoryArea::WorkingRam);
    OracleState {
        label: label.to_string(),
        frame_index,
        game_mode: read_u16(memory.as_ref(), GAME_MODE_ADDR),
        current_player: read_u8(memory.as_ref(), CURRENT_PLAYER_ADDR),
        number_hitpoints: read_u16(memory.as_ref(), NUMBER_HITPOINTS_ADDR),
        current_helmet_id: read_u16(memory.as_ref(), CURRENT_HELMET_ID_ADDR),
        current_level_id: read_u16(memory.as_ref(), CURRENT_LEVEL_ID_ADDR),
        player_1_level_id: read_u16(memory.as_ref(), PLAYER_1_LEVEL_ID_ADDR),
        camera_x_pos: read_u32(memory.as_ref(), CAMERA_X_POS_ADDR),
        camera_y_pos: read_u32(memory.as_ref(), CAMERA_Y_POS_ADDR),
        kid_x_pos: read_u32(memory.as_ref(), KID_X_POS_ADDR),
        kid_y_pos: read_u32(memory.as_ref(), KID_Y_POS_ADDR),
        number_objects: read_u16(memory.as_ref(), NUMBER_OBJECTS_ADDR),
        number_gfx_objects: read_u16(memory.as_ref(), NUMBER_GFX_OBJECTS_ADDR),
        active_enemies: capture_enemy_states(memory.as_ref()),
    }
}

fn capture_enemy_states(
    memory: &dyn jgenesis_common::debug::DebugMemoryView,
) -> Vec<OracleEnemyState> {
    let mut enemies = Vec::new();
    for slot_index in 0..GFX_OBJECT_SLOT_COUNT {
        let base = GFX_OBJECT_RAM_ADDR + slot_index * GFX_OBJECT_SLOT_SIZE;
        if read_u8(memory, base + DISPLAY_FLAG_OFFSET) == 0 {
            continue;
        }
        let object_type = read_u16(memory, base + OBJECT_META_OFFSET) & 0x7F;
        let Ok(object_type) = u8::try_from(object_type) else {
            continue;
        };
        if !matches!(object_type, 0x01 | 0x1C) {
            continue;
        }
        enemies.push(OracleEnemyState {
            object_type,
            x_pos: read_u32(memory, base + OBJECT_X_POS_OFFSET),
            y_pos: read_u32(memory, base + OBJECT_Y_POS_OFFSET),
            x_velocity: read_i32(memory, base + OBJECT_X_VEL_OFFSET),
            y_velocity: read_i32(memory, base + OBJECT_Y_VEL_OFFSET),
            facing: if read_u8(memory, base + X_DIRECTION_OFFSET) == 0 {
                EnemyFacing::Left
            } else {
                EnemyFacing::Right
            },
        });
    }
    enemies.sort_by_key(|enemy| (enemy.object_type, enemy.x_pos, enemy.y_pos));
    enemies
}

fn work_ram_index(address: usize) -> usize {
    address & 0xFFFF
}

fn read_u8(memory: &dyn jgenesis_common::debug::DebugMemoryView, address: usize) -> u8 {
    memory.read(work_ram_index(address))
}

fn read_u16(memory: &dyn jgenesis_common::debug::DebugMemoryView, address: usize) -> u16 {
    u16::from_be_bytes([read_u8(memory, address), read_u8(memory, address + 1)])
}

fn read_u32(memory: &dyn jgenesis_common::debug::DebugMemoryView, address: usize) -> u32 {
    (u32::from(read_u16(memory, address)) << 16) | u32::from(read_u16(memory, address + 2))
}

fn read_i32(memory: &dyn jgenesis_common::debug::DebugMemoryView, address: usize) -> i32 {
    read_u32(memory, address) as i32
}

fn apply_single_frames(script: &mut [u8], frames: &[usize], buttons: u8) {
    for frame in frames {
        if let Some(slot) = script.get_mut(*frame) {
            *slot |= buttons;
        }
    }
}

fn apply_range(script: &mut [u8], start: usize, end: usize, buttons: u8) {
    for slot in script
        .iter_mut()
        .take(end.saturating_add(1))
        .skip(start)
    {
        *slot |= buttons;
    }
}

fn inputs_from_mask(mask: u8) -> GenesisInputs {
    let p1 = GenesisJoypadState {
        up: mask & (1 << 0) != 0,
        down: mask & (1 << 1) != 0,
        left: mask & (1 << 2) != 0,
        right: mask & (1 << 3) != 0,
        b: mask & (1 << 4) != 0,
        c: mask & (1 << 5) != 0,
        a: mask & (1 << 6) != 0,
        start: mask & (1 << 7) != 0,
        ..GenesisJoypadState::default()
    };
    GenesisInputs {
        p1,
        p2: GenesisJoypadState::default(),
    }
}

fn write_png(path: &Path, width: u32, height: u32, rgb: &[u8]) -> Result<()> {
    image::save_buffer_with_format(path, rgb, width, height, ColorType::Rgb8, ImageFormat::Png)
        .with_context(|| format!("failed to write {}", path.display()))
}

fn write_wav(path: &Path, samples: &[f32]) -> Result<()> {
    let spec = WavSpec {
        channels: 2,
        sample_rate: SAMPLE_RATE as u32,
        bits_per_sample: 32,
        sample_format: SampleFormat::Float,
    };
    let mut writer = WavWriter::create(path, spec)
        .with_context(|| format!("failed to create {}", path.display()))?;
    for sample in samples {
        writer
            .write_sample(*sample)
            .with_context(|| format!("failed to write samples to {}", path.display()))?;
    }
    writer
        .finalize()
        .with_context(|| format!("failed to finalise {}", path.display()))
}

fn sha1_hex(bytes: &[u8]) -> String {
    let mut hasher = Sha1::new();
    hasher.update(bytes);
    let digest = hasher.finalize();
    let mut hex = String::with_capacity(digest.len() * 2);
    for byte in digest {
        hex.push_str(&format!("{byte:02x}"));
    }
    hex
}
