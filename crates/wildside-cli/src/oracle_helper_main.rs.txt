fn parse_memory_watches(spec: &str) -> Result<Vec<MemoryWatch>> {
    if spec.is_empty() {
        return Ok(Vec::new());
    }
    let mut watches = Vec::new();
    for item in spec.split(',') {
        let (addr_str, type_str) = item.split_once(':').ok_or_else(|| {
            anyhow!("memory watch must be ADDRESS:TYPE, got {item}")
        })?;
        let address = usize::from_str_radix(addr_str.trim_start_matches("0x").trim_start_matches("0X"), 16)
            .with_context(|| format!("invalid address in memory watch: {addr_str}"))?;
        let watch_type = match type_str.to_ascii_lowercase().as_str() {
            "u8" => WatchType::U8,
            "u16" => WatchType::U16,
            "u32" => WatchType::U32,
            "i32" => WatchType::I32,
            other => return Err(anyhow!("unknown watch type {other}; use u8/u16/u32/i32")),
        };
        let label = format!("{addr_str}:{type_str}");
        watches.push(MemoryWatch { address, watch_type, label });
    }
    Ok(watches)
}

fn parse_vdp_pixel_samples(spec: &str) -> Result<Vec<VdpPixelSample>> {
    if spec.is_empty() {
        return Ok(Vec::new());
    }
    spec.split(',')
        .map(|item| {
            let (x, y) = item
                .split_once('/')
                .ok_or_else(|| anyhow!("VDP pixel sample must be X/Y, got {item}"))?;
            Ok(VdpPixelSample {
                x: x.trim().parse().with_context(|| format!("invalid sample x: {x}"))?,
                y: y.trim().parse().with_context(|| format!("invalid sample y: {y}"))?,
            })
        })
        .collect()
}

fn read_watch_value(work_ram: &[u8], watch: &MemoryWatch) -> i64 {
    match watch.watch_type {
        WatchType::U8 => i64::from(read_wram_u8(work_ram, watch.address)),
        WatchType::U16 => i64::from(read_wram_u16(work_ram, watch.address)),
        WatchType::U32 => i64::from(read_wram_u32(work_ram, watch.address)),
        WatchType::I32 => i64::from(read_wram_i32(work_ram, watch.address)),
    }
}

fn parse_frame_list(spec: &str) -> Result<Vec<usize>> {
    if spec.is_empty() {
        return Ok(Vec::new());
    }
    spec.split(',')
        .map(|s| s.trim().parse::<usize>().with_context(|| format!("invalid frame number: {s}")))
        .collect()
}

#[derive(Default)]
struct CaptureRenderer {
    width: u32,
    height: u32,
    rgb: Vec<u8>,
}

impl Renderer for CaptureRenderer {
    type Err = io::Error;

    fn render_frame(
        &mut self,
        frame_buffer: &[Color],
        frame_size: FrameSize,
        _: RenderFrameOptions,
    ) -> Result<(), Self::Err> {
        self.width = frame_size.width;
        self.height = frame_size.height;
        self.rgb.clear();
        self.rgb.reserve_exact(frame_size.width as usize * frame_size.height as usize * 3);
        for color in frame_buffer
            .iter()
            .take(frame_size.width as usize * frame_size.height as usize)
        {
            self.rgb.extend_from_slice(&[color.r, color.g, color.b]);
        }
        Ok(())
    }
}

#[derive(Default)]
struct AudioCollector {
    samples: Vec<f32>,
}

impl AudioOutput for AudioCollector {
    type Err = io::Error;

    fn push_sample(&mut self, sample_l: f64, sample_r: f64) -> Result<(), Self::Err> {
        self.samples.push(sample_l as f32);
        self.samples.push(sample_r as f32);
        Ok(())
    }
}

fn main() -> Result<()> {
    let mut args = env::args().skip(1);
    let mut rom_path = None;
    let mut out_dir = None;
    let mut scenario_slug = None;
    let mut scenario_file = None;
    let mut mode = OutputMode::Capture;
    let mut skip_frames_before: Option<usize> = None;
    let mut memory_watches_spec = String::new();
    let mut wram_dump_frames_spec = String::new();
    let mut vram_dump_frames_spec = String::new();
    let mut cram_dump_frames_spec = String::new();
    let mut vsram_dump_frames_spec = String::new();
    let mut vdp_pixel_samples_spec = String::new();
    let mut capture_cpu_state = false;
    let mut capture_settle_trace = false;
    let mut capture_rng_stream = false;

    while let Some(arg) = args.next() {
        match arg.as_str() {
            "--rom" => rom_path = args.next().map(PathBuf::from),
            "--out-dir" => out_dir = args.next().map(PathBuf::from),
            "--scenario" => scenario_slug = args.next(),
            "--scenario-file" => scenario_file = args.next().map(PathBuf::from),
            "--mode" => mode = parse_mode(&args.next().context("missing --mode value")?)?,
            "--skip-frames-before" => skip_frames_before = args.next().and_then(|s| s.parse().ok()),
            "--memory-watches" => memory_watches_spec = args.next().unwrap_or_default(),
            "--wram-dump-frames" => wram_dump_frames_spec = args.next().unwrap_or_default(),
            "--vram-dump-frames" => vram_dump_frames_spec = args.next().unwrap_or_default(),
            "--cram-dump-frames" => cram_dump_frames_spec = args.next().unwrap_or_default(),
            "--vsram-dump-frames" => vsram_dump_frames_spec = args.next().unwrap_or_default(),
            "--vdp-pixel-samples" => vdp_pixel_samples_spec = args.next().unwrap_or_default(),
            "--capture-cpu-state" => capture_cpu_state = true,
            "--capture-settle-trace" => capture_settle_trace = true,
            "--capture-rng-stream" => capture_rng_stream = true,
            other => return Err(anyhow!("unknown arg {other}")),
        }
    }

    let rom_path = rom_path.context("missing --rom")?;
    let out_dir = out_dir.context("missing --out-dir")?;
    fs::create_dir_all(&out_dir)
        .with_context(|| format!("failed to create {}", out_dir.display()))?;

    let resolved = resolve_scenario(scenario_slug.as_deref(), scenario_file.as_deref())?;
    let memory_watches = parse_memory_watches(&memory_watches_spec)?;
    let wram_dump_frames = parse_frame_list(&wram_dump_frames_spec)?;
    let vram_dump_frames = parse_frame_list(&vram_dump_frames_spec)?;
    let cram_dump_frames = parse_frame_list(&cram_dump_frames_spec)?;
    let vsram_dump_frames = parse_frame_list(&vsram_dump_frames_spec)?;
    let vdp_pixel_samples = parse_vdp_pixel_samples(&vdp_pixel_samples_spec)?;

    let rom = fs::read(&rom_path)
        .with_context(|| format!("failed to read ROM from {}", rom_path.display()))?;
    match mode {
        OutputMode::Capture => {
            run_capture(&rom, &out_dir, &resolved, skip_frames_before, &memory_watches, &wram_dump_frames, &vram_dump_frames, &cram_dump_frames, &vsram_dump_frames, &vdp_pixel_samples, capture_cpu_state, capture_settle_trace, capture_rng_stream)?;
        }
        OutputMode::EnemyLayoutTrace => {
            run_enemy_layout_trace(&rom, &resolved)?;
        }
    }
    Ok(())
}

fn resolve_scenario(
    slug: Option<&str>,
    file_path: Option<&Path>,
) -> Result<ResolvedScenario> {
    match (slug, file_path) {
        (Some(_), Some(_)) => Err(anyhow!("pass --scenario or --scenario-file, not both")),
        (None, None) => Err(anyhow!("missing --scenario or --scenario-file")),
        (Some(slug), None) => {
            let scenario = parse_scenario(slug)?;
            Ok(ResolvedScenario {
                slug: scenario_slug(scenario).to_string(),
                script: scenario_script(scenario),
                frame_count: scenario_frame_count(scenario),
                checkpoints: scenario_checkpoints(scenario).to_vec(),
                _owned_labels: Vec::new(),
                warp_to_level: None,
                start_position: None,
                start_velocity: None,
                camera_position: None,
                playtest_form: None,
                initial_diamond_count: None,
                initial_lives: None,
                settle_oracle_warp: false,
            })
        }
        (None, Some(path)) => {
            let json = fs::read_to_string(path)
                .with_context(|| format!("failed to read scenario file {}", path.display()))?;
            let sf: ScenarioFile = serde_json::from_str(&json)
                .with_context(|| format!("failed to parse scenario file {}", path.display()))?;
            // Build owned FrameCheckpoints — labels are leaked to 'static for API compat.
            // This is fine: one allocation per run, lives for the process lifetime.
            let owned_labels: Vec<String> = sf.checkpoints.iter().map(|cp| cp.label.clone()).collect();
            let checkpoints: Vec<FrameCheckpoint> = sf.checkpoints.iter().enumerate().map(|(i, cp)| {
                FrameCheckpoint {
                    label: Box::leak(owned_labels[i].clone().into_boxed_str()),
                    frame_index: cp.frame_index,
                }
            }).collect();
            let settle_oracle_warp = sf.settle_oracle_warp
                || sf.slug.starts_with("idle_level_full_frames_")
                || sf.slug.ends_with("_settled_recording_oracle");
            Ok(ResolvedScenario {
                slug: sf.slug,
                script: sf.script,
                frame_count: sf.frame_count,
                checkpoints,
                _owned_labels: owned_labels,
                warp_to_level: sf.warp_to_level,
                start_position: sf.start_position,
                start_velocity: sf.start_velocity,
                camera_position: sf.camera_position,
                playtest_form: sf.playtest_form,
                initial_diamond_count: sf.initial_diamond_count,
                initial_lives: sf.initial_lives,
                settle_oracle_warp,
            })
        }
    }
}

fn run_capture(rom: &[u8], out_dir: &Path, resolved: &ResolvedScenario, skip_frames_before: Option<usize>, memory_watches: &[MemoryWatch], wram_dump_frames: &[usize], vram_dump_frames: &[usize], cram_dump_frames: &[usize], vsram_dump_frames: &[usize], vdp_pixel_samples: &[VdpPixelSample], capture_cpu_state: bool, capture_settle_trace: bool, capture_rng_stream: bool) -> Result<()> {
    let mut checkpoint_index = 0usize;
    let mut save_writer = NullSaveWriter;
    let mut emulator = GenesisEmulator::create(rom.to_vec(), genesis_config(), &mut save_writer);
    emulator.update_audio_output_frequency(SAMPLE_RATE);
    let mut renderer = CaptureRenderer::default();
    let mut audio = AudioCollector::default();
    let mut frame_index = 0usize;
    let mut rendered_frames = Vec::new();
    let mut captured_states = Vec::new();
    let mut frame_crcs: Vec<u32> = Vec::with_capacity(resolved.frame_count);
    let mut frame_rng_states = capture_rng_stream
        .then(|| Vec::with_capacity(resolved.frame_count))
        .unwrap_or_default();
    let mut all_memory_watches: Vec<HashMap<String, i64>> = Vec::new();
    let mut all_vdp_pixel_samples: Vec<OracleVdpPixelSampleRow> = Vec::new();
    let mut settle_trace: Vec<OracleSettleTraceRow> = Vec::new();

    // Pre-warp: boot to level 0, then advance to target level.
    let force_target_level_for_probe = should_force_target_level_for_probe(resolved);
    let align_direct_probe_input_latch = resolved.warp_to_level.is_some()
        && force_target_level_for_probe
        && !resolved.settle_oracle_warp;

    if let Some(target_level) = resolved.warp_to_level {
        let settle_trace_out = capture_settle_trace.then_some(&mut settle_trace);
        boot_and_warp_oracle(
            &mut emulator,
            &mut renderer,
            &mut audio,
            &mut save_writer,
            target_level,
            resolved.settle_oracle_warp,
            force_target_level_for_probe,
            settle_trace_out,
        )?;
    }
    if let Some(position) = resolved.start_position {
        apply_probe_start_position(&mut emulator, position);
    }
    if let Some(velocity) = resolved.start_velocity {
        apply_probe_start_velocity(&mut emulator, velocity);
    }
    if let Some(position) = resolved.camera_position {
        apply_probe_camera_position(&mut emulator, position);
    }
    if let Some(helmet_id) = resolved.playtest_form {
        apply_probe_playtest_form(&mut emulator, helmet_id)?;
    }
    if let Some(count) = resolved.initial_diamond_count {
        apply_probe_diamond_count(&mut emulator, count);
    }
    if let Some(count) = resolved.initial_lives {
        apply_probe_lives_count(&mut emulator, count);
    }

    while frame_index < resolved.frame_count {
        let pre_frame_eol_vdp_pixel_samples_debug = resolved
            .checkpoints
            .get(checkpoint_index)
            .filter(|checkpoint| checkpoint.frame_index == frame_index)
            .and_then(|checkpoint| {
                matches!(checkpoint.label, "flag_touch" | "eol_reveal" | "eol_counter" | "eol_tail")
                    .then(|| eol_vdp_pixel_samples_debug_loc_c070(&mut emulator, None))
            });
        if resolved.playtest_form.is_some() {
            clear_probe_transform_residue(&mut emulator);
        }
        let inputs = inputs_from_mask(resolved.script.get(frame_index).copied().unwrap_or(0));
        loop {
            if emulator.tick(&mut renderer, &mut audio, &inputs, &mut save_writer)? ==
                TickEffect::FrameRendered
            {
                if resolved.playtest_form.is_some() {
                    clear_probe_transform_residue(&mut emulator);
                }
                while checkpoint_index < resolved.checkpoints.len()
                    && resolved.checkpoints[checkpoint_index].frame_index == frame_index
                {
                    let should_capture = skip_frames_before.is_none_or(|skip| frame_index >= skip);
                    if should_capture {
                        let checkpoint = resolved.checkpoints[checkpoint_index];
                        let png_path = out_dir.join(format!(
                            "{:04}_{}.png",
                            checkpoint.frame_index, checkpoint.label
                        ));
                        write_png(&png_path, renderer.width, renderer.height, &renderer.rgb)?;
                        rendered_frames.push(OracleFrame {
                            label: checkpoint.label.to_string(),
                            frame_index: checkpoint.frame_index,
                            sha1: sha1_hex(&renderer.rgb),
                            png_path,
                        });
                        captured_states.push(capture_state(
                            &mut emulator,
                            checkpoint.label,
                            checkpoint.frame_index,
                            capture_cpu_state,
                            Some((renderer.width, renderer.height, &renderer.rgb)),
                            pre_frame_eol_vdp_pixel_samples_debug.as_deref(),
                        ));
                    }
                    checkpoint_index += 1;
                }
                // Per-frame CRC of authoritative RAM state
                {
                    let work_ram = capture_memory(&mut emulator, GenesisMemoryArea::WorkingRam, 0x1_0000);
                    if capture_rng_stream {
                        let offset = read_wram_u16(&work_ram, RNG_OFFSET_ADDR);
                        let ring = work_ram
                            [work_ram_index(RNG_RAM_START_ADDR)..work_ram_index(RNG_RAM_END_ADDR)]
                            .try_into()
                            .expect("RNG_RAM bounds must contain exactly 55 bytes");
                        frame_rng_states.push(
                            wildside_core::RomRandomFrameState::from_initialised_parts(
                                offset,
                                wildside_core::RomRandomRing::from_bytes(ring),
                            ),
                        );
                    }
                    let mut crc_hasher = crc32fast::Hasher::new();
                    crc_hasher.update(&read_wram_u16(&work_ram, GAME_MODE_ADDR).to_le_bytes());
                    crc_hasher.update(&read_wram_u16(&work_ram, CURRENT_LEVEL_ID_ADDR).to_le_bytes());
                    crc_hasher.update(&read_wram_u32(&work_ram, CAMERA_X_POS_ADDR).to_le_bytes());
                    crc_hasher.update(&read_wram_u32(&work_ram, CAMERA_Y_POS_ADDR).to_le_bytes());
                    crc_hasher.update(&read_wram_u32(&work_ram, KID_X_POS_ADDR).to_le_bytes());
                    crc_hasher.update(&read_wram_u32(&work_ram, KID_Y_POS_ADDR).to_le_bytes());
                    // Velocity excluded from CRC: capture-timing differs between
                    // native (post-frame) and oracle (VBlank). Positions are ground truth.
                    crc_hasher.update(&read_wram_u16(&work_ram, CURRENT_HELMET_ID_ADDR).to_le_bytes());
                    crc_hasher.update(&read_wram_u32(&work_ram, SCORE_ADDR).to_le_bytes());
                    crc_hasher.update(&read_wram_u16(&work_ram, NUMBER_DIAMONDS_ADDR).to_le_bytes());
                    crc_hasher.update(&[read_wram_u8(&work_ram, DIAMOND_POWER_ACTIVE_ADDR)]);
                    crc_hasher.update(&read_wram_u8(&work_ram, CURRENT_PLAYER_ADDR).to_le_bytes());
                    // player_1_level_id excluded: demo-mode timing artefact.
                    frame_crcs.push(crc_hasher.finalize());

                    // Memory watches: read requested addresses from WRAM
                    if !memory_watches.is_empty() {
                        let mut watch_map = HashMap::new();
                        watch_map.insert("frame".to_string(), frame_index as i64);
                        for watch in memory_watches {
                            watch_map.insert(watch.label.clone(), read_watch_value(&work_ram, watch));
                        }
                        all_memory_watches.push(watch_map);
                    }

                    // WRAM dump: write full 64KB to file at requested frames
                    if wram_dump_frames.contains(&frame_index) {
                        write_memory_dump(
                            out_dir,
                            "wram",
                            frame_index,
                            MemoryDumpArea::Wram,
                            &mut emulator,
                        )?;
                    }
                    if vram_dump_frames.contains(&frame_index) {
                        write_memory_dump(
                            out_dir,
                            "vram",
                            frame_index,
                            MemoryDumpArea::Vram,
                            &mut emulator,
                        )?;
                    }
                    if cram_dump_frames.contains(&frame_index) {
                        write_memory_dump(
                            out_dir,
                            "cram",
                            frame_index,
                            MemoryDumpArea::Cram,
                            &mut emulator,
                        )?;
                    }
                    if vsram_dump_frames.contains(&frame_index) {
                        write_memory_dump(
                            out_dir,
                            "vsram",
                            frame_index,
                            MemoryDumpArea::Vsram,
                            &mut emulator,
                        )?;
                    }
                    if !vdp_pixel_samples.is_empty() {
                        all_vdp_pixel_samples.push(OracleVdpPixelSampleRow {
                            frame: frame_index,
                            samples: vdp_pixel_samples_debug(
                                &mut emulator,
                                vdp_pixel_samples,
                                Some((renderer.width, renderer.height, &renderer.rgb)),
                            ),
                        });
                    }
                }
                if align_direct_probe_input_latch && frame_index == 0 {
                    seed_input_latches(
                        &mut emulator,
                        resolved.script.first().copied().unwrap_or(0),
                    );
                }
                frame_index += 1;
                break;
            }
        }
    }

    let audio_path = out_dir.join("audio.wav");
    write_wav(&audio_path, &audio.samples)?;
    let manifest = OracleManifest {
        sample_rate: SAMPLE_RATE as u32,
        audio_path,
        frames: rendered_frames,
        states: captured_states,
        frame_crcs,
        frame_rng_states,
        memory_watches: all_memory_watches,
        vdp_pixel_samples: all_vdp_pixel_samples,
        settle_trace,
    };
    println!("{}", serde_json::to_string_pretty(&manifest)?);
    Ok(())
}

fn write_memory_dump(
    out_dir: &Path,
    prefix: &str,
    frame_index: usize,
    area: MemoryDumpArea,
    emulator: &mut GenesisEmulator,
) -> Result<()> {
    let memory_area = match area {
        MemoryDumpArea::Wram => GenesisMemoryArea::WorkingRam,
        MemoryDumpArea::Vram => GenesisMemoryArea::Vram,
        MemoryDumpArea::Cram => GenesisMemoryArea::Cram,
        MemoryDumpArea::Vsram => GenesisMemoryArea::Vsram,
    };
    let len = match area {
        MemoryDumpArea::Wram | MemoryDumpArea::Vram => 0x1_0000,
        MemoryDumpArea::Cram => 0x80,
        MemoryDumpArea::Vsram => 0x50,
    };
    let bytes = capture_memory(emulator, memory_area, len);
    let dump_path = out_dir.join(format!("{prefix}_frame_{frame_index}.bin"));
    fs::write(&dump_path, &bytes)
        .with_context(|| format!("failed to write {}", dump_path.display()))?;
    Ok(())
}

fn run_enemy_layout_trace(rom: &[u8], resolved: &ResolvedScenario) -> Result<()> {
    let mut checkpoint_index = 0usize;
    let mut save_writer = NullSaveWriter;
    let mut emulator = GenesisEmulator::create(rom.to_vec(), genesis_config(), &mut save_writer);
    emulator.update_audio_output_frequency(SAMPLE_RATE);
    let mut renderer = CaptureRenderer::default();
    let mut audio = AudioCollector::default();
    let mut frame_index = 0usize;
    let mut checkpoints_out = Vec::new();

    let force_target_level_for_probe = should_force_target_level_for_probe(resolved);

    if let Some(target_level) = resolved.warp_to_level {
        boot_and_warp_oracle(
            &mut emulator,
            &mut renderer,
            &mut audio,
            &mut save_writer,
            target_level,
            resolved.settle_oracle_warp,
            force_target_level_for_probe,
            None,
        )?;
    }

    while frame_index < resolved.frame_count {
        let inputs = inputs_from_mask(resolved.script.get(frame_index).copied().unwrap_or(0));
        loop {
            if emulator.tick(&mut renderer, &mut audio, &inputs, &mut save_writer)? ==
                TickEffect::FrameRendered
            {
                while checkpoint_index < resolved.checkpoints.len()
                    && resolved.checkpoints[checkpoint_index].frame_index == frame_index
                {
                    let checkpoint = resolved.checkpoints[checkpoint_index];
                    checkpoints_out.push(capture_enemy_layout_checkpoint(
                        &mut emulator,
                        checkpoint.label,
                        checkpoint.frame_index,
                    ));
                    checkpoint_index += 1;
                }
                frame_index += 1;
                break;
            }
        }
    }

    let manifest = OracleEnemyLayoutTraceManifest {
        scenario: resolved.slug.clone(),
        checkpoints: checkpoints_out,
    };
    println!("{}", serde_json::to_string_pretty(&manifest)?);
    Ok(())
}

fn parse_scenario(value: &str) -> Result<Scenario> {
    match value {
        "boot_title_idle_10s" => Ok(Scenario::BootTitleIdle10s),
        "boot_title_start_level1" => Ok(Scenario::BootTitleStartLevel1),
        "boot_title_start_level1_move_right_5s" => Ok(Scenario::BootTitleStartLevel1MoveRight5s),
        "level1_enemy_window" => Ok(Scenario::Level1EnemyWindow),
        "walk_right_3s" => Ok(Scenario::WalkRight3s),
        "jump_from_standstill" => Ok(Scenario::JumpFromStandstill),
        _ => Err(anyhow!("unknown scenario {value}")),
    }
}

fn parse_mode(value: &str) -> Result<OutputMode> {

        _ => Err(anyhow!("unknown mode {value}")),
    }
}

fn scenario_script(scenario: Scenario) -> Vec<u8> {
    let mut script = vec![0; scenario_frame_count(scenario).saturating_add(1)];
    match scenario {
        Scenario::BootTitleIdle10s => {}
        Scenario::BootTitleStartLevel1 => {
            apply_single_frames(&mut script, &wildside_core::BOOT_TO_LEVEL_START_FRAMES, 0x80);
        }
        Scenario::BootTitleStartLevel1MoveRight5s => {
            apply_single_frames(&mut script, &wildside_core::BOOT_TO_LEVEL_START_FRAMES, 0x80);
            apply_range(&mut script, 720, 1_019, 1 << 3);
        }
        Scenario::Level1EnemyWindow => {
            apply_single_frames(&mut script, &wildside_core::BOOT_TO_LEVEL_START_FRAMES, 0x80);
            apply_range(&mut script, 720, 899, 1 << 3);
        }
        Scenario::WalkRight3s => {
            apply_single_frames(&mut script, &wildside_core::BOOT_TO_LEVEL_START_FRAMES, 0x80);
            apply_range(&mut script, 720, 899, 1 << 3);
        }
        Scenario::JumpFromStandstill => {
            apply_single_frames(&mut script, &wildside_core::BOOT_TO_LEVEL_START_FRAMES, 0x80);
            apply_range(&mut script, 720, 723, 1 << 4);
        }
    }
    script
}

fn capture_state(
    emulator: &mut GenesisEmulator,
    label: &str,
    frame_index: usize,
    capture_cpu_state: bool,
    rendered_frame: Option<(u32, u32, &[u8])>,
    pre_frame_eol_vdp_pixel_samples_debug: Option<&str>,
) -> OracleState {
    let cpu_regs = if capture_cpu_state {
        let debug = emulator.debug();
        Some(CpuRegisters {
            pc: debug.m68k_pc(),
            data_registers: debug.m68k_data_registers(),
            address_registers: debug.m68k_address_registers(),
            status_register: debug.m68k_status_register(),
        })
    } else {
        None
    };
    let work_ram = capture_memory(emulator, GenesisMemoryArea::WorkingRam, 0x1_0000);
    let eol_vdp_debug = matches!(label, "flag_touch" | "eol_reveal" | "eol_counter" | "eol_tail")
        .then(|| eol_vdp_registers_debug_loc_c070(emulator));
    let eol_vram_plane_rows_debug =
        matches!(label, "flag_touch" | "eol_reveal" | "eol_counter" | "eol_tail")
            .then(|| eol_vram_plane_rows_debug_loc_c070(emulator));
    let eol_scroll_debug = matches!(label, "flag_touch" | "eol_reveal" | "eol_counter" | "eol_tail")
        .then(|| eol_scroll_words_debug_loc_c070(emulator));
    let eol_cram_debug = matches!(label, "flag_touch" | "eol_reveal" | "eol_counter" | "eol_tail")
        .then(|| eol_cram_words_debug_loc_c070(emulator));
    let eol_vdp_pixel_samples_debug =
        matches!(label, "flag_touch" | "eol_reveal" | "eol_counter" | "eol_tail")
            .then(|| eol_vdp_pixel_samples_debug_loc_c070(emulator, rendered_frame));
    let game_mode = read_wram_u16(&work_ram, GAME_MODE_ADDR);
    let currently_transforming = read_wram_u8(&work_ram, CURRENTLY_TRANSFORMING_ADDR) != 0;
    let game_paused = read_wram_u8(&work_ram, GAME_PAUSED_ADDR) != 0;
    let pause_menu_option = read_wram_u8(&work_ram, PAUSE_OPTION_ADDR);
    // ASM variables.asm: Options_Suboption_Speed = $FFFFFDC9 and
    // Options_Suboption_Controls = $FFFFFDCA. These are demo-skipped
    // gameplay inputs for #305 interactive-lane probes.
    let options_speed_button_fast = read_wram_u8(&work_ram, OPTIONS_SUBOPTION_SPEED_ADDR) != 0;
    let options_controls_scheme = read_wram_u16(&work_ram, OPTIONS_SUBOPTION_CONTROLS_ADDR);
    let mut gameplay_debug_fields = BTreeMap::new();
    if currently_transforming {
        gameplay_debug_fields.insert("currently_transforming".into(), "true".into());
        gameplay_debug_fields.insert("gameplay_hibernated".into(), "true".into());
    }
    if game_paused {
        gameplay_debug_fields.insert("game_paused".into(), "true".into());
        gameplay_debug_fields.insert("pause_menu_option".into(), pause_menu_option.to_string());
    }
    if options_controls_scheme != 0 {
        gameplay_debug_fields.insert(
            "options_controls_scheme".into(),
            options_controls_scheme.to_string(),
        );
    }
    if options_speed_button_fast {
        gameplay_debug_fields.insert("options_speed_button_fast".into(), "true".into());
    }
    if matches!(label, "flag_touch" | "eol_reveal" | "eol_counter" | "eol_tail") {
        gameplay_debug_fields.insert(
            "eol_visible_gfx_slots_loc_d468".into(),
            eol_visible_gfx_slots_debug_loc_d468(&work_ram),
        );
        gameplay_debug_fields.insert(
            "eol_sprite_table_entries_loc_c070".into(),
            eol_sprite_table_entries_debug_loc_c070(emulator),
        );
        if let Some(eol_vram_plane_rows_debug) = eol_vram_plane_rows_debug {
            gameplay_debug_fields.insert(
                "eol_vram_plane_rows_loc_c070".into(),
                eol_vram_plane_rows_debug,
            );
        }
        if let Some(eol_vdp_debug) = eol_vdp_debug {
            gameplay_debug_fields.insert("eol_vdp_registers_loc_c070".into(), eol_vdp_debug);
        }
        if let Some(eol_scroll_debug) = eol_scroll_debug {
            gameplay_debug_fields.insert("eol_scroll_words_loc_c070".into(), eol_scroll_debug);
        }
        if let Some(eol_cram_debug) = eol_cram_debug {
            gameplay_debug_fields.insert("eol_cram_words_loc_c070".into(), eol_cram_debug);
        }
        if let Some(eol_vdp_pixel_samples_debug) = eol_vdp_pixel_samples_debug {
            gameplay_debug_fields.insert(
                "eol_vdp_pixel_samples_loc_c070".into(),
                eol_vdp_pixel_samples_debug,
            );
        }
        if let Some(pre_frame_eol_vdp_pixel_samples_debug) =
            pre_frame_eol_vdp_pixel_samples_debug
        {
            gameplay_debug_fields.insert(
                "eol_pre_frame_vdp_pixel_samples_loc_c070".into(),
                pre_frame_eol_vdp_pixel_samples_debug.to_string(),
            );
        }
    }
    let kid_collision_type =
        read_wram_u16(&work_ram, GFX_OBJECT_RAM_ADDR + OBJECT_COLLISION_TYPE_OFFSET);
    let kid_collision_source =
        read_wram_u16(&work_ram, GFX_OBJECT_RAM_ADDR + OBJECT_COLLISION_SOURCE_OFFSET);
    let background_plane_b_rows_debug = matches!(game_mode, 0x0008 | 0x000C | 0x0010)
        .then(|| background_plane_b_rows_debug_loc_307d8(emulator));
    let background_vdp_pixel_samples_debug = matches!(game_mode, 0x0008 | 0x000C | 0x0010)
        .then(|| background_vdp_pixel_samples_debug_loc_c070(emulator, rendered_frame));
    let rom = emulator.debug().memory_view(GenesisMemoryArea::CartridgeRom);
    if matches!(game_mode, 0x0008 | 0x000C | 0x0010) {
        gameplay_debug_fields.insert(
            "kid_collision_type".into(),
            format_collision_type(kid_collision_type),
        );
        gameplay_debug_fields.insert(
            "kid_collision_source".into(),
            format!("{kid_collision_source:#06x}"),
        );
        if let Some(platform_debug) = platform_collision_scratch_debug_loc_257e(&work_ram) {
            gameplay_debug_fields.insert(
                "platform_collision_scratch_loc_257e".into(),
                platform_debug,
            );
        }
        let platform_slots_debug = oracle_platform_slots_debug_loc_257e(&work_ram);
        if !platform_slots_debug.is_empty() {
            gameplay_debug_fields.insert(
                "oracle_platform_slots_loc_257e".into(),
                platform_slots_debug,
            );
        }
        let shooter_debug = shooter_object_slots_debug_loc_ec22(&work_ram);
        if !shooter_debug.is_empty() {
            gameplay_debug_fields.insert("oracle_shooter_slots_loc_ec22".into(), shooter_debug);
        }
        let object_debug = object_slots_debug_loc_e1e(&work_ram, rom.as_ref());
        if !object_debug.is_empty() {
            gameplay_debug_fields.insert("oracle_object_slots_loc_e1e".into(), object_debug);
        }
        gameplay_debug_fields.insert(
            "oracle_background_scroll_loc_3050a".into(),
            background_scroll_debug_loc_3050a(&work_ram),
        );
        gameplay_debug_fields.insert(
            "oracle_background_plane_b_rows_loc_307d8".into(),
            background_plane_b_rows_debug
                .expect("in-game background Plane-B row debug should be captured"),
        );
        gameplay_debug_fields.insert(
            "oracle_background_vdp_pixel_samples_loc_c070".into(),
            background_vdp_pixel_samples_debug
                .expect("in-game background VDP pixel samples should be captured"),
        );
        let drips_object_debug = drips_object_slots_debug_loc_33488(&work_ram);
        if !drips_object_debug.is_empty() {

        }
    }
    OracleState {
        label: label.to_string(),
        frame_index,
        game_mode,
        boot_phase: boot_phase_label(game_mode).map(str::to_string),
        boot_phase_elapsed: None,

            .then(|| read_wram_u8(&work_ram, MAIN_MENU_OPTION_ADDR)),
        current_player: read_wram_u8(&work_ram, CURRENT_PLAYER_ADDR),
        number_hitpoints: read_wram_u16(&work_ram, NUMBER_HITPOINTS_ADDR),
        current_helmet_id: read_wram_u16(&work_ram, CURRENT_HELMET_ID_ADDR),
        current_level_id: read_wram_u16(&work_ram, CURRENT_LEVEL_ID_ADDR),
        player_1_level_id: read_wram_u16(&work_ram, PLAYER_1_LEVEL_ID_ADDR),
        camera_x_pos: read_wram_u32(&work_ram, CAMERA_X_POS_ADDR),
        camera_y_pos: read_wram_u32(&work_ram, CAMERA_Y_POS_ADDR),
        kid_x_pos: read_wram_u32(&work_ram, KID_X_POS_ADDR),
        kid_y_pos: read_wram_u32(&work_ram, KID_Y_POS_ADDR),
        number_objects: read_wram_u16(&work_ram, NUMBER_OBJECTS_ADDR),
        number_gfx_objects: read_wram_u16(&work_ram, NUMBER_GFX_OBJECTS_ADDR),
        game_paused,
        pause_menu_option,

        score: read_wram_u32(&work_ram, SCORE_ADDR),
        number_lives: read_wram_u16(&work_ram, NUMBER_LIVES_ADDR),
        number_diamonds: read_wram_u16(&work_ram, NUMBER_DIAMONDS_ADDR),
        diamond_power_active: read_wram_u8(&work_ram, DIAMOND_POWER_ACTIVE_ADDR) != 0,
        murder_wall_x: if read_wram_u8(&work_ram, MURDER_WALL_FLAG_ADDR) != 0 {
            Some(read_wram_i32(&work_ram, MURDER_WALL_X_POS_ADDR))
        } else {
            None
        },
        kid_x_vel: if matches!(game_mode, 0x0008 | 0x000C | 0x0010) {
            // ASM GfxObject offsets `x_vel=$26`/`y_vel=$2A` live on Kid's
            // actual gfx slot (`Addr_GfxObject_Kid=$FFFFF85E`), which is
            // $FFFF24E0 after settled Load_InGame. The older fixed base slot
            // at $FFFF2448 stays zero and masks live movement in TAS probes.
            Some(read_wram_i32(&work_ram, KID_X_VEL_ADDR))
        } else {
            None
        },
        kid_y_vel: if matches!(game_mode, 0x0008 | 0x000C | 0x0010) {
            Some(read_wram_i32(&work_ram, KID_Y_VEL_ADDR))
        } else {
            None
        },
        character_movement: Some(read_wram_u16(&work_ram, CHARACTER_MOVEMENT_ADDR)),
        just_received_damage: Some(read_wram_u8(&work_ram, JUST_RECEIVED_DAMAGE_ADDR) != 0),
        kid_collision_type: matches!(game_mode, 0x0008 | 0x000C | 0x0010)
            .then_some(kid_collision_type),
        kid_collision_source: matches!(game_mode, 0x0008 | 0x000C | 0x0010)
            .then_some(kid_collision_source),
        kid_x_direction: matches!(game_mode, 0x0008 | 0x000C | 0x0010)
            .then_some(read_wram_u8(&work_ram, KID_X_DIRECTION_ADDR)),
        invulnerable_frames: Some(read_wram_u8(&work_ram, INVULNERABLE_FRAMES_ADDR)),
        currently_transforming: Some(currently_transforming),
        rng_offset: Some(read_wram_u16(&work_ram, RNG_OFFSET_ADDR)),
        rng_last_byte: Some(rng_last_byte(&work_ram)),
        rng_ring: Some(
            work_ram[work_ram_index(RNG_RAM_START_ADDR)..work_ram_index(RNG_RAM_END_ADDR)]
                .to_vec(),
        ),
        level_special_effects: Some(read_wram_u16(&work_ram, LEVEL_SPECIAL_EFFECTS_ADDR)),
        gameplay_debug_fields,
        cpu_registers: cpu_regs,
    }
}

fn boot_phase_label(game_mode: u16) -> Option<&'static str> {
    match game_mode {
        0x0000 => Some("sega_screen"),
        0x0004 => Some("intro_sequence_1"),
        0x0018 => Some("intro_sequence_2"),
        0x001C => Some("intro_sequence_3"),
        0x0020 => Some("intro_sequence_4"),
        0x0024 => Some("intro_sequence_5"),
        0x002C => Some("title_screen"),
        0x0014 => Some("options_menu"),
        0x0008 => Some("starting_level"),
        0x0010 => Some("demo_play"),
        0x0030 => Some("ending_sequence"),
        _ => None,
    }
}

fn capture_enemy_states(
    work_ram: &[u8],
    rom: &dyn jgenesis_common::debug::DebugMemoryView,
) -> Vec<OracleEnemyState> {
    let mut enemies = Vec::new();
    let layout_object_ptr = read_wram_u32(work_ram, ENEMY_LAYOUT_OBJECT_PTR_ADDR) as usize;
    let layout_object_count = read_rom_u8(rom, layout_object_ptr + 1);
    let kid_gfx_object = read_wram_u32(work_ram, ADDR_GFX_OBJECT_KID_ADDR) as usize;
    for (slot_index, base) in active_gfx_object_slots_loc_14a4(work_ram) {
        if read_wram_u8(work_ram, base + DISPLAY_FLAG_OFFSET) == 0 {
            continue;
        }
        // ASM GfxObjects_CollisionKid loc_1CF2 copies an enemy's `$3A`
        // collision source into Kid's GfxObject. The source identifies what
        // Kid touched, not Kid's object type, so never use it as the direct
        // enemy fallback for the slot named by Addr_GfxObject_Kid.
        if base == kid_gfx_object {
            continue;
        }
        let parent_object = read_wram_u32(work_ram, base + PARENT_OBJECT_OFFSET) as usize;
        if !(OBJECT_RAM_ADDR..OBJECT_RAM_END_ADDR).contains(&parent_object) {
            continue;
        }
        let collision_type = read_wram_u16(work_ram, base + OBJECT_COLLISION_TYPE_OFFSET);
        let collision_source = read_wram_u16(work_ram, base + OBJECT_COLLISION_SOURCE_OFFSET);
        let layout_ptr = read_wram_u32(work_ram, parent_object + OBJECT_LAYOUT_PTR_OFFSET) as usize;
        let layout_index = layout_index_from_ptr(layout_object_ptr, layout_object_count, layout_ptr);
        let object_type = if let Some(_layout_index) = layout_index {
            read_rom_u8(rom, layout_ptr)
        } else if let Some(object_type) =
            direct_enemy_object_type_from_collision_source(collision_source)
        {
            object_type
        } else {
            continue;
        };
        let mut debug_fields = BTreeMap::new();
        if collision_type != 0 || collision_source != 0 {
            debug_fields.insert("collision_type".into(), format_collision_type(collision_type));
            debug_fields.insert("collision_source".into(), format!("{collision_source:#06x}"));
        }
        if layout_index.is_none() && collision_source != 0 {
            debug_fields.insert(
                "object_type_source".into(),
                "collision_source_direct_loc_32146".into(),
            );
        }
        enemies.push(OracleEnemyState {
            object_type,
            x_pos: read_wram_u32(work_ram, base + OBJECT_X_POS_OFFSET),
            y_pos: read_wram_u32(work_ram, base + OBJECT_Y_POS_OFFSET),
            x_velocity: read_wram_i32(work_ram, base + OBJECT_X_VEL_OFFSET),
            y_velocity: read_wram_i32(work_ram, base + OBJECT_Y_VEL_OFFSET),
            facing: if read_wram_u8(work_ram, base + X_DIRECTION_OFFSET) == 0 {
                EnemyFacing::Left
            } else {
                EnemyFacing::Right
            },
            collision_type: Some(collision_type),
            collision_source: Some(collision_source),
            layout_index,
            layout_ptr: layout_index.map(|_layout_index| layout_ptr as u32),
            parent_object_ptr: Some(parent_object as u32),
            gfx_slot_index: Some(slot_index),
            on_ground: None,
            current_hp: Some(read_wram_u16(work_ram, base + GFX_OBJECT_CURRENT_HP_OFFSET)),
            debug_fields,
        });
    }
    enemies.sort_by_key(|enemy| (enemy.object_type, enemy.x_pos, enemy.y_pos));
    enemies
}

fn active_gfx_object_slots_loc_14a4(work_ram: &[u8]) -> Vec<(usize, usize)> {
    // ASM Make_SpritesFromGfxObjects loc_14A4 follows Addr_FirstGfxObjectSlot's
    // linked list. Freed physical slots retain stale fields until allocation,
    // but they are not active GfxObjects and must not enter oracle state.
    let mut slots = Vec::new();
    let mut seen = [false; GFX_OBJECT_SLOT_COUNT];
    let mut base =
        read_wram_u32(work_ram, ADDR_FIRST_GFX_OBJECT_SLOT_ADDR) as usize;
    for _ in 0..GFX_OBJECT_SLOT_COUNT {
        let Some(offset) = base.checked_sub(GFX_OBJECT_RAM_ADDR) else {
            break;
        };
        if offset >= GFX_OBJECT_SLOT_SIZE * GFX_OBJECT_SLOT_COUNT
            || offset % GFX_OBJECT_SLOT_SIZE != 0
        {
            break;
        }
        let slot_index = offset / GFX_OBJECT_SLOT_SIZE;
        if seen[slot_index] {
            break;
        }
        seen[slot_index] = true;
        slots.push((slot_index, base));
        base = read_wram_u32(work_ram, base) as usize;
    }
    slots
}

fn platform_collision_scratch_debug_loc_257e(work_ram: &[u8]) -> Option<String> {
    // ASM Platforms_CheckCollision loc_257E zeroes FAA2-FAA5 before the
    // loc_2672/loc_2686/loc_26A2/loc_26B6/loc_26D4/loc_26FE probes write
    // side/top/bottom contact scratch and FA96's jump-platform reference.
    let faa2 = read_wram_u8(work_ram, PLATFORM_COLLISION_SCRATCH_FAA2_ADDR) != 0;
    let faa3 = read_wram_u8(work_ram, PLATFORM_COLLISION_SCRATCH_FAA3_ADDR) != 0;
    let faa4 = read_wram_u8(work_ram, PLATFORM_COLLISION_SCRATCH_FAA4_ADDR) != 0;
    let faa5 = read_wram_u8(work_ram, PLATFORM_COLLISION_SCRATCH_FAA5_ADDR) != 0;
    let fa96 = read_wram_u16(work_ram, PLATFORM_COLLISION_JUMP_PLATFORM_REF_FA96_ADDR);
    if !(faa2 || faa3 || faa4 || faa5 || fa96 != 0) {
        return None;
    }
    let fa96 = if fa96 == 0 {
        String::from("none")
    } else {
        format!("{fa96:#06x}")
    };
    Some(format!(
        "FAA2={} FAA3={} FAA4={} FAA5={} FA96={fa96}",
        u8::from(faa2),
        u8::from(faa3),
        u8::from(faa4),
        u8::from(faa5)
    ))
}

fn oracle_platform_slots_debug_loc_257e(work_ram: &[u8]) -> String {
    // ASM Platforms_CheckCollision loc_257E and Move_Platforms sub_23EA both
    // walk Addr_FirstPlatformSlot's `$22`-byte Platform_RAM linked list.
    let platform_ram_start = work_ram_index(PLATFORM_RAM_ADDR);
    let platform_ram_end = platform_ram_start + PLATFORM_SLOT_SIZE * PLATFORM_SLOT_COUNT;
    let number_platforms = read_wram_u16(work_ram, NUMBER_PLATFORMS_ADDR);
    let mut slot_addr = usize::from(read_wram_u16(work_ram, ADDR_FIRST_PLATFORM_SLOT_ADDR));
    let mut guard = 0usize;
    let mut slots = Vec::new();
    while slot_addr != 0
        && (platform_ram_start..platform_ram_end).contains(&slot_addr)
        && guard < PLATFORM_SLOT_COUNT
    {
        let next = read_wram_u16(work_ram, slot_addr);
        let x_pos = read_wram_u32(work_ram, slot_addr + 2);
        let y_pos = read_wram_u32(work_ram, slot_addr + 6);
        let x_vel = read_wram_i32(work_ram, slot_addr + 0x0A);
        let y_vel = read_wram_i32(work_ram, slot_addr + 0x0E);
        let state = read_wram_u32(work_ram, slot_addr + 0x12);
        let timer = read_wram_u16(work_ram, slot_addr + 0x16);
        let art = read_wram_u16(work_ram, slot_addr + 0x18);
        let width = read_wram_u16(work_ram, slot_addr + 0x1A);
        let height = read_wram_u16(work_ram, slot_addr + 0x1C);
        let render = read_wram_u8(work_ram, slot_addr + 0x1F);
        let layout_offset = read_wram_u16(work_ram, slot_addr + 0x20);
        slots.push(format!(
            "0xffff{slot_addr:04x}:next={next:#06x},xy={x_pos:#010x},{y_pos:#010x},vel={x_vel:#010x},{y_vel:#010x},state={state:#010x},timer={timer:#06x},art={art:#06x},size={width}x{height},render={render:#04x},off={layout_offset:#06x}"
        ));
        slot_addr = usize::from(next);
        guard += 1;
    }
    if slots.is_empty() {
        String::new()
    } else {
        format!("count={number_platforms};{}", slots.join(";"))
    }
}

fn shooter_object_slots_debug_loc_ec22(work_ram: &[u8]) -> String {
    // ASM LoadBulletsFromEvanescentBlock/Process_ShooterObjects
    // (kid.asm:19080-19620) threads the `$C`-byte ShooterObject_RAM records
    // through Addr_FirstShooterObjectSlot. Keep this oracle-side dump narrow:
    // #214 only needs to distinguish shooter-slot scheduling from later
    // block-destroy worker lifetime.
    let mut slots = Vec::new();
    let mut slot_addr = usize::from(read_wram_u16(
        work_ram,
        ADDR_FIRST_SHOOTER_OBJECT_SLOT_ADDR,
    ));
    let mut guard = 0usize;
    while slot_addr != 0
        && (SHOOTER_OBJECT_RAM_ADDR..SHOOTER_OBJECT_RAM_END_ADDR).contains(&slot_addr)
        && guard < SHOOTER_OBJECT_SLOT_COUNT
    {
        let source = read_wram_u16(work_ram, slot_addr);
        let x_px = read_wram_u16(work_ram, slot_addr + 2);
        let y_px = read_wram_u16(work_ram, slot_addr + 4);
        let direction = read_wram_u16(work_ram, slot_addr + 6);
        let counter = read_wram_u16(work_ram, slot_addr + 8);
        let next = read_wram_u16(work_ram, slot_addr + 0x0A);
        slots.push(format!(
            "{slot_addr:#06x}:src={source:#06x},x={x_px},y={y_px},dir={direction},ctr={counter:#06x},next={next:#06x}"
        ));
        slot_addr = usize::from(next);
        guard += 1;
    }
    slots.join(";")
}

fn object_slots_debug_loc_e1e(
    work_ram: &[u8],
    rom: &dyn jgenesis_common::debug::DebugMemoryView,
) -> String {
    // ASM Execute_Objects loc_E1E (kid.asm:1636): the active object list is
    // threaded from Addr_FirstObjectSlot through the longword at offset 0.
    // Dump code pointer, coroutine registers, hibernate counter, and owned gfx
    // position. Object+$16..$36 stores d0-d3/a0-a3 across hibernation; the
    // motion registers and fixed-point gfx state make enemy scheduler/RNG and
    // collision timing auditable without adding family-specific probes.
    let mut slots = Vec::new();
    let layout_object_ptr = read_wram_u32(work_ram, ENEMY_LAYOUT_OBJECT_PTR_ADDR) as usize;
    let layout_object_count = read_rom_u8(rom, layout_object_ptr + 1);
    let mut object_addr = read_wram_u32(work_ram, ADDR_FIRST_OBJECT_SLOT_ADDR) as usize;
    let mut guard = 0usize;
    while object_addr != 0
        && (OBJECT_RAM_ADDR..OBJECT_RAM_END_ADDR).contains(&object_addr)
        && guard < 0x32
    {
        let code = read_wram_u32(work_ram, object_addr + 4);
        let hibernate = read_wram_u16(work_ram, object_addr + 8);
        let d0 = read_wram_u32(work_ram, object_addr + 0x16);
        let d1 = read_wram_u32(work_ram, object_addr + 0x1A);
        let d2 = read_wram_u32(work_ram, object_addr + 0x1E);
        let d3 = read_wram_u32(work_ram, object_addr + 0x22);
        let object_44 = read_wram_u16(work_ram, object_addr + 0x44);
        let object_46 = read_wram_u16(work_ram, object_addr + 0x46);
        let object_48 = read_wram_u16(work_ram, object_addr + 0x48);
        let object_4a = read_wram_u16(work_ram, object_addr + 0x4A);
        let object_4c = read_wram_u16(work_ram, object_addr + 0x4C);
        let object_50 = read_wram_u16(work_ram, object_addr + 0x50);
        let layout_ptr = read_wram_u32(work_ram, object_addr + OBJECT_LAYOUT_PTR_OFFSET) as usize;
        let layout = layout_index_from_ptr(layout_object_ptr, layout_object_count, layout_ptr)
            .map(|layout_index| {
                let object_type = read_rom_u8(rom, layout_ptr);
                format!(",li={layout_index},type={object_type:#04x},lp={layout_ptr:#08x}")
            })
            .unwrap_or_default();
        let gfx_addr = read_wram_u32(work_ram, object_addr + 0x36) as usize;
        let gfx = if (GFX_OBJECT_RAM_ADDR
            ..GFX_OBJECT_RAM_ADDR + GFX_OBJECT_SLOT_SIZE * GFX_OBJECT_SLOT_COUNT)
            .contains(&gfx_addr)
        {
            // ASM Dragon_Flying.asm loc_34664 stores the visible GfxObject at
            // Object+$36 and loc_3478C can keep the Object live while the
            // helper's active-enemy GfxObject scan omits display-cleared slots.
            format!(
                "@{:#010x},{:#010x}:vel={:#010x},{:#010x},spr={:#06x},disp={:#04x},col={},src={:#06x}",
                read_wram_u32(work_ram, gfx_addr + OBJECT_X_POS_OFFSET),
                read_wram_u32(work_ram, gfx_addr + OBJECT_Y_POS_OFFSET),
                read_wram_u32(work_ram, gfx_addr + 0x26),
                read_wram_u32(work_ram, gfx_addr + 0x2A),
                read_wram_u16(work_ram, gfx_addr + 0x22),
                read_wram_u8(work_ram, gfx_addr + DISPLAY_FLAG_OFFSET),
                format_collision_type(read_wram_u16(
                    work_ram,
                    gfx_addr + OBJECT_COLLISION_TYPE_OFFSET
                )),
                read_wram_u16(work_ram, gfx_addr + OBJECT_COLLISION_SOURCE_OFFSET),
            )
        } else {
            String::from("@-")
        };
        slots.push(format!(

fn background_source_window_debug_loc_30744(work_ram: &[u8]) -> String {
    // ASM sub_30744 (kid.asm:34563) reads source bytes from the live
    // `$FFFF87B2` buffer via `(a0, Horiz_Scroll_Data.col)`, where `a0` starts
    // at `$FFFF4D5C[Camera_Y_pos>>5]` and advances by `Background_width`.
    // Dump the source columns that cover the #289 B18 cloud-tail writes and
    // nearby ocean-row investigation without serialising the whole buffer.
    const VDP_ROWS: [usize; 2] = [18, 29];
    const FIRST_COLUMN: usize = 0x28;
    const LAST_COLUMN: usize = 0x38;

    let camera_y = read_wram_u32(work_ram, CAMERA_Y_POS_ADDR) >> 16;
    let camera_chunk = (camera_y >> 5) as usize;
    let background_width = usize::from(read_wram_u16(work_ram, BACKGROUND_WIDTH_ADDR));
    let row_base = usize::from(read_wram_u16(
        work_ram,
        BACKGROUND_ROW_OFFSETS_ADDR + camera_chunk * 2,
    ));

    VDP_ROWS
        .into_iter()
        .filter_map(|vdp_row| {
            let row_delta = (vdp_row + 0x20 - (camera_chunk & 0x1F)) & 0x1F;
            if row_delta > 0x1C {
                return None;
            }
            let source_offset = row_base + row_delta * background_width;
            let bytes = (FIRST_COLUMN..=LAST_COLUMN)
                .map(|column| {
                    format!(
                        "{:02X}",
                        read_wram_u8(work_ram, 0xFF_0000 + source_offset + column)
                    )
                })
                .collect::<Vec<_>>()
                .join(",");
            Some(format!(
                "B{vdp_row:02}@{source_offset:04X}[{FIRST_COLUMN:02X}..{LAST_COLUMN:02X}]={bytes}"
            ))
        })
        .collect::<Vec<_>>()
        .join(";")
}

fn background_plane_b_spans_debug_loc_307d8(emulator: &mut GenesisEmulator) -> String {
    const PLANE_B_BASE_LOC_307D8: usize = 0xE000;
    const PLANE_WIDTH_CELLS: usize = 64;
    const ROWS: [usize; 9] = [8, 17, 18, 19, 22, 23, 27, 29, 30];

    // ASM sub_307D8 streams a compact row span, then sub_30744 appends
    // single exposed columns. Summarise non-baseline spans so #289 settle
    // traces show the row-history shape without parsing the full 64 words.
    let vram = emulator.debug().memory_view(GenesisMemoryArea::Vram);
    ROWS
        .into_iter()
        .map(|row| {
            let mut spans = Vec::new();
            let mut span_start: Option<usize> = None;
            let mut span_word = 0u16;
            for col in 0..PLANE_WIDTH_CELLS {
                let cell_offset = PLANE_B_BASE_LOC_307D8 + (row * PLANE_WIDTH_CELLS + col) * 2;
                let word = read_rom_u16(&*vram, cell_offset);
                let is_baseline = matches!(word, 0x0780 | 0x0094);
                match (span_start, is_baseline, word == span_word) {
                    (None, false, _) => {
                        span_start = Some(col);
                        span_word = word;
                    }
                    (Some(start), true, _) => {
                        spans.push(format_span_loc_307d8(start, col - 1, span_word));
                        span_start = None;
                    }
                    (Some(start), false, false) => {
                        spans.push(format_span_loc_307d8(start, col - 1, span_word));
                        span_start = Some(col);
                        span_word = word;
                    }
                    _ => {}
                }
            }
            if let Some(start) = span_start {
                spans.push(format_span_loc_307d8(start, PLANE_WIDTH_CELLS - 1, span_word));
            }
            let spans = if spans.is_empty() {
                "none".to_owned()
            } else {
                spans.join("|")
            };
            format!("B{row:02}={spans}")
        })
        .collect::<Vec<_>>()
        .join(";")
}

fn layout_index_from_ptr(
    layout_object_ptr: usize,
    layout_object_count: u8,
    layout_ptr: usize,
) -> Option<usize> {
    if layout_object_ptr == 0 || layout_ptr < layout_object_ptr + 2 {
        return None;
    }
    let object_table_end = layout_object_ptr + 2 + usize::from(layout_object_count) * 8;
    if layout_ptr >= object_table_end {
        return None;
    }
    let relative_offset = layout_ptr - (layout_object_ptr + 2);
    (relative_offset % 8 == 0).then_some(relative_offset / 8)
}

fn direct_enemy_object_type_from_collision_source(collision_source: u16) -> Option<u8> {
    // ASM Dragon_Flying.asm loc_34664/loc_34BAC writes object_meta `$C`
    // before calling loc_32146, which copies the direct enemy id into the
    // GfxObject collision source. Some live dragon Object states later clear
    // or repurpose Object+$44, so the enemy-layout pointer path above cannot
    // identify them even though the displayed GfxObject is still authoritative.
    let object_type = u8::try_from(collision_source).ok()?;
    // ASM constants.asm starts non-enemy GfxObject metadata at
    // `objid_prizediamond=$64`; PrizeHelmetCyclone_Init writes
    // `objid_cyclone=$70` before loc_104AE. Those values share the same
    // GfxObject field as loc_32146's direct enemy id, so accepting every
    // non-zero byte misclassifies prizes as enemies.
    const FIRST_NON_ENEMY_OBJECT_META: u8 = 0x64;
    (object_type != 0 && object_type < FIRST_NON_ENEMY_OBJECT_META).then_some(object_type)
}

fn format_collision_type(collision_type: u16) -> String {
    let label = match collision_type {
        0x0000 => "empty",
        0x0004 => "rightwall",
        0x0008 => "leftwall",
        0x000C => "floor",
        0x0010 => "ceiling",
        0x0014 => "slopeup",
        0x0018 => "slopedown",
        0x001C => "hurt",
        0x0020 => "kidright",
        0x0024 => "kidleft",
        0x0028 => "kidbelow",
        0x002C => "kidabove",
        0xFFFF => "special",
        _ => "unknown",
    };
    format!("{collision_type:#06x}({label})")
}

fn capture_enemy_layout_checkpoint(
    emulator: &mut GenesisEmulator,
    label: &str,
    frame_index: usize,
) -> OracleEnemyLayoutCheckpoint {
    let work_ram = capture_memory(emulator, GenesisMemoryArea::WorkingRam, 0x1_0000);
    let rom = emulator.debug().memory_view(GenesisMemoryArea::CartridgeRom);
    let header_ptr = read_wram_u32(&work_ram, ENEMY_LAYOUT_HEADER_PTR_ADDR);
    let object_ptr = read_wram_u32(&work_ram, ENEMY_LAYOUT_OBJECT_PTR_ADDR);
    let slot0 = read_rom_u16(rom.as_ref(), header_ptr as usize + 2);
    let slot1 = read_rom_u16(rom.as_ref(), header_ptr as usize + 4);
    let slot2 = read_rom_u16(rom.as_ref(), header_ptr as usize + 6);
    let enemy_type_slots = vec![slot0, slot1, slot2];
    let object_batch_flags = read_rom_u8(rom.as_ref(), object_ptr as usize);
    let object_count = read_rom_u8(rom.as_ref(), object_ptr as usize + 1);
    let mut objects = Vec::with_capacity(object_count as usize);
    for object_index in 0..usize::from(object_count) {
        let base = object_ptr as usize + 2 + object_index * 8;
        let object_type = read_rom_u8(rom.as_ref(), base);
        objects.push(OracleEnemyLayoutObject {
            object_type,
            resolved_object_type: resolve_object_type(object_type, &enemy_type_slots),
            flags: read_rom_u8(rom.as_ref(), base + 1),
            hit_points_plus_one: read_rom_u16(rom.as_ref(), base + 2),
            x: read_rom_u16(rom.as_ref(), base + 4),
            y: read_rom_u16(rom.as_ref(), base + 6),
        });
    }

    OracleEnemyLayoutCheckpoint {
        label: label.to_string(),
        frame_index,
        camera_x_pos: read_wram_u32(&work_ram, CAMERA_X_POS_ADDR),
        camera_y_pos: read_wram_u32(&work_ram, CAMERA_Y_POS_ADDR),
        header_ptr,
        object_ptr,
        axis_uses_y_position: read_rom_u16(rom.as_ref(), header_ptr as usize) != 0,
        enemy_type_slots,
        upper_bound: read_rom_u16(rom.as_ref(), header_ptr as usize + 8),
        object_batch_flags,
        object_count,
        objects,
        visible_enemies: capture_visible_enemy_layouts(&work_ram, rom.as_ref()),
    }
}

fn capture_visible_enemy_layouts(
    work_ram: &[u8],
    rom: &dyn jgenesis_common::debug::DebugMemoryView,
) -> Vec<OracleVisibleEnemyLayout> {
    let mut visible_enemies = Vec::new();
    for slot_index in 0..GFX_OBJECT_SLOT_COUNT {
        let base = GFX_OBJECT_RAM_ADDR + slot_index * GFX_OBJECT_SLOT_SIZE;
        if read_wram_u8(work_ram, base + DISPLAY_FLAG_OFFSET) == 0 {
            continue;
        }
        let parent_object_ptr = read_wram_u32(work_ram, base + PARENT_OBJECT_OFFSET);
        let parent_object = parent_object_ptr as usize;
        if !(OBJECT_RAM_ADDR..OBJECT_RAM_END_ADDR).contains(&parent_object) {
            continue;
        }
        let layout_ptr = read_wram_u32(work_ram, parent_object + OBJECT_LAYOUT_PTR_OFFSET);
        if layout_ptr == 0 {
            continue;
        }
        visible_enemies.push(OracleVisibleEnemyLayout {
            object_type: read_rom_u8(rom, layout_ptr as usize),
            layout_ptr,
            parent_object_ptr,
            x_pos: read_wram_u32(work_ram, base + OBJECT_X_POS_OFFSET),
            y_pos: read_wram_u32(work_ram, base + OBJECT_Y_POS_OFFSET),
        });
    }
    visible_enemies.sort_by_key(|enemy| (enemy.object_type, enemy.layout_ptr, enemy.x_pos));
    visible_enemies
}

fn eol_visible_gfx_slots_debug_loc_d468(work_ram: &[u8]) -> String {
    let mut slots = Vec::new();
    for slot_index in 0..GFX_OBJECT_SLOT_COUNT {
        let base = GFX_OBJECT_RAM_ADDR + slot_index * GFX_OBJECT_SLOT_SIZE;
        if read_wram_u8(work_ram, base + DISPLAY_FLAG_OFFSET) == 0 {
            continue;
        }
        let sprite = read_wram_u16(work_ram, base + GFX_OBJECT_SPRITE_OFFSET);
        let vram_tile = read_wram_u16(work_ram, base + GFX_OBJECT_VRAM_TILE_OFFSET);
        if sprite == 0 && vram_tile != 0x0517 && vram_tile != 0x0409 {
            continue;
        }
        slots.push(format!(

            read_wram_u16(work_ram, base + OBJECT_Y_POS_OFFSET),
            read_wram_u8(work_ram, base + 0x11),
            read_wram_u8(work_ram, base + 0x10),

}

fn eol_sprite_table_entries_debug_loc_c070(emulator: &mut GenesisEmulator) -> String {
    const SPRITE_TABLE_BASE_LOC_C070: usize = 0x1000;

    let vram = capture_memory(emulator, GenesisMemoryArea::Vram, 0x1_0000);
    let mut entries = Vec::new();
    let mut sprite_index = 0usize;
    let mut visited = [false; 80];
    for ordinal in 0..80 {
        if sprite_index >= visited.len() || visited[sprite_index] {
            break;
        }
        visited[sprite_index] = true;

        let entry_base = SPRITE_TABLE_BASE_LOC_C070 + sprite_index * 8;
        let y_word = read_wram_u16(&vram, entry_base);
        let size_link = read_wram_u16(&vram, entry_base + 2);
        let pattern = read_wram_u16(&vram, entry_base + 4);
        let x_word = read_wram_u16(&vram, entry_base + 6);
        let x_pos = i32::from(x_word) - 128;
        let y_pos = i32::from(y_word) - 128;
        let width_tiles = ((size_link >> 10) & 0x03) + 1;
        let height_tiles = ((size_link >> 8) & 0x03) + 1;
        entries.push(format!(
            "#{sprite_index:02X}/ord={ordinal}:x={x_pos}/y={y_pos},raw={x_word:04X}/{y_word:04X},size={width_tiles}x{height_tiles},link={:02X},pat={pattern:04X},tile={:03X},pal={},pri={},hf={},vf={}",

            pattern & 0x07FF,
            (pattern >> 13) & 0x03,
            if pattern & 0x8000 != 0 { 1 } else { 0 },
            if pattern & 0x0800 != 0 { 1 } else { 0 },
            if pattern & 0x1000 != 0 { 1 } else { 0 },
        ));

        let next_link = usize::from(size_link & 0x7F);
        if next_link == 0 {
            break;
        }
        sprite_index = next_link;
    }

    entries.join(";")
}

fn eol_vdp_registers_debug_loc_c070(emulator: &mut GenesisEmulator) -> String {
    let mut registers = Vec::new();
    emulator
        .debug()
        .dump_vdp_registers(|register_name, fields| {
            if !matches!(
                register_name,
                "Register #1" | "Register #2" | "Register #4" | "Register #11" | "Register #13"
                    | "Register #16"
            ) {
                return;
            }
            let fields = fields
                .iter()
                .map(|(key, value)| format!("{key}={value}"))
                .collect::<Vec<_>>()
                .join(",");
            registers.push(format!("{register_name}:{fields}"));
        });
    registers.join(";")
}

fn eol_vram_plane_rows_debug_loc_c070(emulator: &mut GenesisEmulator) -> String {
    const PLANE_A_BASE_LOC_C070: usize = 0x0000;
    const PLANE_B_BASE_LOC_C070: usize = 0xE000;
    const PLANE_WIDTH_CELLS: usize = 64;
    const SCREEN_WIDTH_CELLS: usize = 40;
    const ROWS: [usize; 7] = [0, 1, 2, 7, 13, 15, 25];

    let vram = emulator.debug().memory_view(GenesisMemoryArea::Vram);
    let mut rows = Vec::new();
    for row in ROWS {
        // ASM loc_C070 sets Plane A=$0000 and Plane B=$E000 before the flag-touch hibernate.
        rows.push(format!(
            "A{row:02}={}",
            eol_vram_plane_row_debug(&*vram, PLANE_A_BASE_LOC_C070, row)
        ));
        rows.push(format!(
            "B{row:02}={}",
            eol_vram_plane_row_debug(&*vram, PLANE_B_BASE_LOC_C070, row)
        ));
    }

    fn eol_vram_plane_row_debug(
        vram: &dyn jgenesis_common::debug::DebugMemoryView,
        plane_base: usize,
        row: usize,
    ) -> String {
        (0..SCREEN_WIDTH_CELLS)
            .map(|col| {
                let cell_offset = plane_base + (row * PLANE_WIDTH_CELLS + col) * 2;
                format!("{:04X}", read_rom_u16(vram, cell_offset))
            })
            .collect::<Vec<_>>()
            .join(",")
    }

    rows.join(";")
}

fn eol_scroll_words_debug_loc_c070(emulator: &mut GenesisEmulator) -> String {
    const H_SCROLL_TABLE_BASE_LOC_C070: usize = 0x1400;
    const SCANLINES: [usize; 7] = [0, 16, 32, 60, 111, 148, 200];

    let vsram = emulator.debug().memory_view(GenesisMemoryArea::Vsram);
    let vs_a = read_rom_u16(&*vsram, 0);
    let vs_b = read_rom_u16(&*vsram, 2);
    drop(vsram);

    let vram = emulator.debug().memory_view(GenesisMemoryArea::Vram);
    let mut words = vec![format!("vs=A:{vs_a:04X},B:{vs_b:04X}")];
    for scanline in SCANLINES {
        // ASM loc_C070 leaves VDP register #13 at $1400; register #11 selects per-line H-scroll.
        let offset = H_SCROLL_TABLE_BASE_LOC_C070 + scanline * 4;
        words.push(format!(
            "y{scanline:03}=A:{:04X},B:{:04X}",
            read_rom_u16(&*vram, offset),
            read_rom_u16(&*vram, offset + 2)
        ));
    }
    words.join(";")
}

fn rendered_frame_rgb(
    width: u32,
    height: u32,
    rgb: &[u8],
    screen_x: usize,
    screen_y: usize,
) -> Option<[u8; 3]> {
    let width = usize::try_from(width).ok()?;
    let height = usize::try_from(height).ok()?;
    if screen_x >= width || screen_y >= height {
        return None;
    }
    let offset = (screen_y * width + screen_x).checked_mul(3)?;
    Some([*rgb.get(offset)?, *rgb.get(offset + 1)?, *rgb.get(offset + 2)?])
}

fn eol_vdp_sprite_pixel_sample_loc_c070(
    vram: &[u8],
    cram: &[u8],
    sprite_table_base: usize,
    screen_x: usize,
    screen_y: usize,
) -> String {
    let mut sprite_index = 0usize;
    let mut visited = [false; 80];
    for ordinal in 0..80 {
        if sprite_index >= visited.len() || visited[sprite_index] {
            break;
        }
        visited[sprite_index] = true;

        let entry_base = sprite_table_base + sprite_index * 8;
        let y_pos = i32::from(read_wram_u16(vram, entry_base)) - 128;
        let size_link = read_wram_u16(vram, entry_base + 2);
        let pattern = read_wram_u16(vram, entry_base + 4);
        let x_pos = i32::from(read_wram_u16(vram, entry_base + 6)) - 128;
        let width_tiles = usize::from(((size_link >> 10) & 0x03) + 1);
        let height_tiles = usize::from(((size_link >> 8) & 0x03) + 1);
        let width_pixels = (width_tiles * 8) as i32;
        let height_pixels = (height_tiles * 8) as i32;
        let local_x = screen_x as i32 - x_pos;
        let local_y = screen_y as i32 - y_pos;
        if (0..width_pixels).contains(&local_x)
            && (0..height_pixels).contains(&local_y)
            && let Some(sample) = eol_vdp_sprite_entry_pixel_sample_loc_c070(
                vram,
                cram,
                pattern,
                width_tiles,
                height_tiles,
                local_x as usize,
                local_y as usize,
            )
        {
            return format!(
                "#{sprite_index:02X}/ord={ordinal}:xy={x_pos}/{y_pos},size={width_tiles}x{height_tiles},link={:02X},pat={pattern:04X},{sample}",
                size_link & 0x7F,
            );
        }

        let next_link = usize::from(size_link & 0x7F);
        if next_link == 0 {
            break;
        }
        sprite_index = next_link;
    }
    "none".to_string()
}

fn eol_vdp_sprite_entry_pixel_sample_loc_c070(
    vram: &[u8],
    cram: &[u8],
    pattern: u16,
    width_tiles: usize,
    height_tiles: usize,
    local_x: usize,
    local_y: usize,
) -> Option<String> {
    let hflip = pattern & 0x0800 != 0;
    let vflip = pattern & 0x1000 != 0;
    let palette_line = usize::from((pattern >> 13) & 0x03);
    let priority = if pattern & 0x8000 != 0 { 1 } else { 0 };
    let base_tile = usize::from(pattern & 0x07FF);
    let local_x = if hflip {
        width_tiles * 8 - 1 - local_x
    } else {
        local_x
    };
    let local_y = if vflip {
        height_tiles * 8 - 1 - local_y
    } else {
        local_y
    };
    let tile_col = local_x / 8;
    let tile_row = local_y / 8;
    let tile_index = base_tile + tile_col * height_tiles + tile_row;
    let tile_x = local_x & 7;
    let tile_y = local_y & 7;
    let tile_byte = read_wram_u8(vram, tile_index * 32 + tile_y * 4 + tile_x / 2);
    let colour_index = if tile_x & 1 == 0 {
        tile_byte >> 4
    } else {
        tile_byte & 0x0F
    };
    if colour_index == 0 {
        return None;
    }
    let rgb = md_cram_rgb(cram, palette_line * 16 + usize::from(colour_index));
    Some(format!(
        "tile={tile_index:03X},px={tile_x}/{tile_y},pal={palette_line},idx={colour_index:X},pri={priority},rgb={:03}/{:03}/{:03}",
        rgb[0], rgb[1], rgb[2],
    ))
}

fn eol_vdp_plane_pixel_sample(
    vram: &[u8],
    cram: &[u8],
    plane_base: usize,
    plane_width_cells: usize,
    plane_height_cells: usize,
    hscroll: i32,
    vscroll: i32,
    screen_x: usize,
    screen_y: usize,
) -> String {
    let plane_width_pixels = (plane_width_cells * 8) as i32;
    let plane_height_pixels = (plane_height_cells * 8) as i32;
    let world_x = (screen_x as i32 - hscroll).rem_euclid(plane_width_pixels) as usize;
    let world_y = (screen_y as i32 + vscroll).rem_euclid(plane_height_pixels) as usize;
    let cell_x = world_x / 8;
    let cell_y = world_y / 8;
    let entry = read_wram_u16(vram, plane_base + (cell_y * plane_width_cells + cell_x) * 2);
    let tile_x = if entry & 0x0800 != 0 {
        7 - (world_x & 7)
    } else {
        world_x & 7
    };
    let tile_y = if entry & 0x1000 != 0 {
        7 - (world_y & 7)
    } else {
        world_y & 7
    };
    let tile_index = usize::from(entry & 0x07FF);
    let tile_byte = read_wram_u8(vram, tile_index * 32 + tile_y * 4 + tile_x / 2);
    let colour_index = if tile_x & 1 == 0 {
        tile_byte >> 4
    } else {
        tile_byte & 0x0F
    };
    let palette_line = usize::from((entry >> 13) & 0x03);
    let priority = if entry & 0x8000 != 0 { 1 } else { 0 };
    let rgb = if colour_index == 0 {
        "transparent".to_string()
    } else {
        let rgb = md_cram_rgb(cram, palette_line * 16 + usize::from(colour_index));
        format!("rgb={:03}/{:03}/{:03}", rgb[0], rgb[1], rgb[2])
    };
    format!(
        "entry={entry:04X},world={world_x}/{world_y},tile={tile_index:03X},px={tile_x}/{tile_y},pal={palette_line},idx={colour_index:X},pri={priority},{rgb}"
    )
}

fn vdp_plane_pixel_sample_with_tile_row(
    vram: &[u8],
    cram: &[u8],
    plane_base: usize,
    plane_width_cells: usize,
    plane_height_cells: usize,
    hscroll: i32,
    vscroll: i32,
    screen_x: usize,
    screen_y: usize,
) -> String {
    let plane_width_pixels = (plane_width_cells * 8) as i32;
    let plane_height_pixels = (plane_height_cells * 8) as i32;
    let world_x = (screen_x as i32 - hscroll).rem_euclid(plane_width_pixels) as usize;
    let world_y = (screen_y as i32 + vscroll).rem_euclid(plane_height_pixels) as usize;
    let cell_x = world_x / 8;
    let cell_y = world_y / 8;
    let entry = read_wram_u16(vram, plane_base + (cell_y * plane_width_cells + cell_x) * 2);
    let tile_x = if entry & 0x0800 != 0 {
        7 - (world_x & 7)
    } else {
        world_x & 7
    };
    let tile_y = if entry & 0x1000 != 0 {
        7 - (world_y & 7)
    } else {
        world_y & 7
    };
    let tile_index = usize::from(entry & 0x07FF);
    let row_base = tile_index * 32 + tile_y * 4;
    let row_bytes = (0..4)
        .map(|byte_index| read_wram_u8(vram, row_base + byte_index))
        .collect::<Vec<_>>();
    let row_pixels = row_bytes
        .iter()
        .flat_map(|byte| [byte >> 4, byte & 0x0F])
        .map(|pixel| format!("{pixel:X}"))
        .collect::<Vec<_>>()
        .join(",");
    let row_bytes = row_bytes
        .iter()
        .map(|byte| format!("{byte:02X}"))
        .collect::<Vec<_>>()
        .join(",");
    let tile_byte = read_wram_u8(vram, row_base + tile_x / 2);
    let colour_index = if tile_x & 1 == 0 {
        tile_byte >> 4
    } else {
        tile_byte & 0x0F
    };
    let palette_line = usize::from((entry >> 13) & 0x03);
    let priority = if entry & 0x8000 != 0 { 1 } else { 0 };
    let rgb = if colour_index == 0 {
        "transparent".to_string()
    } else {
        let rgb = md_cram_rgb(cram, palette_line * 16 + usize::from(colour_index));
        format!("rgb={:03}/{:03}/{:03}", rgb[0], rgb[1], rgb[2])
    };
    format!(
        "entry={entry:04X},world={world_x}/{world_y},tile={tile_index:03X},px={tile_x}/{tile_y},pal={palette_line},idx={colour_index:X},pri={priority},row={row_pixels},bytes={row_bytes},{rgb}"
    )
}

fn md_cram_rgb(cram: &[u8], palette_index: usize) -> [u8; 3] {
    let colour = read_wram_u16(cram, palette_index * 2);
    let red = ((colour >> 1) & 0x7) as u8;
    let green = ((colour >> 5) & 0x7) as u8;
    let blue = ((colour >> 9) & 0x7) as u8;
    [
        md_scale_channel(red),
        md_scale_channel(green),
        md_scale_channel(blue),
    ]
}

fn md_scale_channel(value: u8) -> u8 {
    ((u16::from(value) * 255 + 3) / 7) as u8
}

fn work_ram_index(address: usize) -> usize {
    address & 0xFFFF
}

fn capture_memory(
    emulator: &mut GenesisEmulator,
    memory_area: GenesisMemoryArea,
    len: usize,
) -> Vec<u8> {
    let memory = emulator.debug().memory_view(memory_area);
    (0..len).map(|index| memory.read(index)).collect()
}

fn read_wram_u8(memory: &[u8], address: usize) -> u8 {
    memory[work_ram_index(address)]
}

fn read_wram_u16(memory: &[u8], address: usize) -> u16 {
    u16::from_be_bytes([read_wram_u8(memory, address), read_wram_u8(memory, address + 1)])
}

fn read_wram_u32(memory: &[u8], address: usize) -> u32 {
    (u32::from(read_wram_u16(memory, address)) << 16)
        | u32::from(read_wram_u16(memory, address + 2))
}

fn rng_last_byte(work_ram: &[u8]) -> u8 {
    let rng_offset = read_wram_u16(work_ram, RNG_OFFSET_ADDR) as i16 as isize;
    read_wram_u8(work_ram, (RNG_RAM_END_ADDR as isize + rng_offset) as usize)
}

fn read_wram_i32(memory: &[u8], address: usize) -> i32 {
    read_wram_u32(memory, address) as i32
}

fn read_rom_u8(memory: &dyn jgenesis_common::debug::DebugMemoryView, address: usize) -> u8 {
    memory.read(address)
}

fn read_rom_u16(memory: &dyn jgenesis_common::debug::DebugMemoryView, address: usize) -> u16 {
    u16::from_be_bytes([read_rom_u8(memory, address), read_rom_u8(memory, address + 1)])
}

fn resolve_object_type(object_type: u8, enemy_type_slots: &[u16]) -> Option<u8> {
    if object_type & 0x80 == 0 {
        return Some(object_type);
    }
    let slot_index = usize::from(object_type & 0x7F);
    let raw_slot = *enemy_type_slots.get(slot_index)?;
    if raw_slot == 0xFFFF {
        return None;
    }
    u8::try_from(raw_slot & 0x3FFF).ok()
}

fn apply_single_frames(script: &mut [u8], frames: &[usize], buttons: u8) {
    for frame in frames {
        if let Some(slot) = script.get_mut(*frame) {
            *slot |= buttons;
        }
    }
}

fn apply_range(script: &mut [u8], start: usize, end: usize, buttons: u8) {
    for slot in script
        .iter_mut()
        .take(end.saturating_add(1))
        .skip(start)
    {
        *slot |= buttons;
    }
}

/// Tick the emulator until one frame is rendered.
fn tick_one_frame(
    emulator: &mut GenesisEmulator,
    renderer: &mut CaptureRenderer,
    audio: &mut AudioCollector,
    inputs: &GenesisInputs,
    save_writer: &mut NullSaveWriter,
) -> Result<()> {
    loop {
        if emulator.tick(renderer, audio, inputs, save_writer)? == TickEffect::FrameRendered {
            return Ok(());
        }
    }
}

/// Boot through menus to level 0, then advance to `target_level` using
/// the LevelSkip_Cheat. Writes the cheat flag directly to work RAM to
/// activate it, then presses Start+A+C to skip one level at a time.
fn boot_and_warp_oracle(
    emulator: &mut GenesisEmulator,
    renderer: &mut CaptureRenderer,
    audio: &mut AudioCollector,
    save_writer: &mut NullSaveWriter,
    target_level: u8,
    settle_oracle_warp: bool,
    force_target_level_for_probe: bool,
    mut settle_trace: Option<&mut Vec<OracleSettleTraceRow>>,
) -> Result<()> {
    let no_input = GenesisInputs::default();

    // Boot: press Start through menus to reach level 0.
    // Same timing as the native boot_to_level.
    for frame in 0..720 {
        let inputs = if wildside_core::BOOT_TO_LEVEL_START_FRAMES.contains(&frame) {
            inputs_from_mask(0x80)
        } else {
            no_input.clone()
        };
        tick_one_frame(emulator, renderer, audio, &inputs, save_writer)?;
        capture_settle_trace_row(
            settle_trace.as_deref_mut(),
            emulator,
            "cold_boot",
            frame + 1,
        );
    }
    capture_settle_trace_row(settle_trace.as_deref_mut(), emulator, "post_boot", 0);

    if target_level == 0 {
        if settle_oracle_warp {
            settle_title_card_target_level(
                emulator,
                renderer,
                audio,
                save_writer,
                target_level,
                settle_trace.as_deref_mut(),
            )?;
        }
        return Ok(());
    }

    // Only warp if target is not level 0 (we're already there after boot).
    if target_level != 0 {
        // Enable the level skip cheat by writing directly to work RAM.
        const LEVEL_SKIP_CHEAT_ADDR: usize = 0xFFFF_FC37;
        {
            let mut mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
            mem.write(work_ram_index(LEVEL_SKIP_CHEAT_ADDR), 0xFF);
        }

        // Set Current_LevelID = target - 1 so that the cheat's addq.w #1
        // increments it to the target.
        {
            let mut mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
            let prev_level = target_level.wrapping_sub(1);
            mem.write(work_ram_index(CURRENT_LEVEL_ID_ADDR), 0x00);
            mem.write(work_ram_index(CURRENT_LEVEL_ID_ADDR) + 1, prev_level);
        }

        // Wait for the player to fully settle after boot before triggering
        // the level skip cheat. The cheat handler runs inside an object's
        // Hibernate loop and needs the player to be in normal gameplay.
        for _ in 0..60 {
            tick_one_frame(emulator, renderer, audio, &no_input, save_writer)?;
        }
        capture_settle_trace_row(settle_trace.as_deref_mut(), emulator, "pre_skip_live", 0);

        // Press Start+A+C for one frame to trigger level skip.
        // Then release and let the game process the level transition.
        let skip_input = GenesisInputs {
            p1: GenesisJoypadState {
                start: true,
                a: true,
                c: true,
                ..GenesisJoypadState::default()
            },
            p2: GenesisJoypadState::default(),
        };
        tick_one_frame(emulator, renderer, audio, &skip_input, save_writer)?;
        if settle_oracle_warp || force_target_level_for_probe {
            clear_input_latches(emulator);
        }
        capture_settle_trace_row(settle_trace.as_deref_mut(), emulator, "post_skip_input", 0);

        // Verify the level skip triggered by checking game_mode.
        // If it changed to 8 (level reload), the skip worked.
        // If not, try again with a longer wait.
        let skip_triggered = {
            let mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
            let gm = u16::from_be_bytes([
                mem.read(work_ram_index(0xFFFF_FBCA)),
                mem.read(work_ram_index(0xFFFF_FBCA) + 1),
            ]);
            gm == 8
        };
        if !skip_triggered {
            // Skip didn't trigger — try pressing again after more idle.
            for _ in 0..60 {
                tick_one_frame(emulator, renderer, audio, &no_input, save_writer)?;
            }
            tick_one_frame(emulator, renderer, audio, &skip_input, save_writer)?;
            if settle_oracle_warp || force_target_level_for_probe {
                clear_input_latches(emulator);
            }

            // Check again.
            let still_not_triggered = {
                let mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
                let gm = u16::from_be_bytes([
                    mem.read(work_ram_index(0xFFFF_FBCA)),
                    mem.read(work_ram_index(0xFFFF_FBCA) + 1),
                ]);
                gm != 8
            };
            if still_not_triggered {
                // Fallback: directly write level ID and force title card.
                // This bypasses the cheat but works for all levels.
                {
                    let mut mem =
                        emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
                    mem.write(work_ram_index(CURRENT_LEVEL_ID_ADDR), 0x00);
                    mem.write(
                        work_ram_index(CURRENT_LEVEL_ID_ADDR) + 1,
                        target_level,
                    );
                    if settle_oracle_warp {
                        mem.write(work_ram_index(PLAYER_1_LEVEL_ID_ADDR), 0x00);
                        mem.write(
                            work_ram_index(PLAYER_1_LEVEL_ID_ADDR) + 1,
                            target_level,
                        );
                    }
                    // Clear transition state and set Game_Mode = 8.
                    mem.write(work_ram_index(0xFFFF_FBCC), 0x00);
                    mem.write(work_ram_index(0xFFFF_FBCC) + 1, 0x00);
                    mem.write(work_ram_index(0xFFFF_FBCE), 0xFF);
                    mem.write(work_ram_index(GAME_MODE_ADDR), 0x00);
                    mem.write(work_ram_index(GAME_MODE_ADDR) + 1, 0x08);
                }
                // Let the title card handler process the mode change.
                for _ in 0..60 {
                    tick_one_frame(
                        emulator, renderer, audio, &no_input, save_writer,
                    )?;
                }
            } else if settle_oracle_warp || force_target_level_for_probe {
                force_warp_level_registers(emulator, target_level);
            }
        } else if settle_oracle_warp || force_target_level_for_probe {
            // Direct probe scenarios patch live WRAM after this bootstrap.
            // Native `boot_to_level` has already selected the requested level,
            // so keep unsettled probe oracles on the same target without
            // changing legacy no-probe state-sweep timing.
            force_warp_level_registers(emulator, target_level);
        }

        // Wait for title card (game_mode=8 → Load_TitleCard → "Press Start").
        // Kept as a fixed 120-frame wait: when we reach this point game_mode
        // is already 0x0008 but the title-card handler needs real frames to
        // finalise its init (VRAM load, sprites, input dispatch) before Start
        // is accepted. Polling game_mode here would break the cheat path,
        // which sets game_mode=0x0008 synchronously and needs these frames.
        for step in 0..120 {
            capture_settle_trace_row(settle_trace.as_deref_mut(), emulator, "title_wait", step);
            tick_one_frame(emulator, renderer, audio, &no_input, save_writer)?;
        }
        capture_settle_trace_row(settle_trace.as_deref_mut(), emulator, "post_title_wait", 0);

        // Press Start to dismiss title card → Load_InGame.
        let start_input = GenesisInputs {
            p1: GenesisJoypadState {
                start: true,
                ..GenesisJoypadState::default()
            },
            p2: GenesisJoypadState::default(),
        };
        tick_one_frame(emulator, renderer, audio, &start_input, save_writer)?;
        if settle_oracle_warp || force_target_level_for_probe {
            clear_input_latches(emulator);
        }
        capture_settle_trace_row(settle_trace.as_deref_mut(), emulator, "post_title_start", 0);
        if force_target_level_for_probe && !settle_oracle_warp {
            drain_direct_probe_title_start_latch(
                emulator,
                renderer,
                audio,
                save_writer,
                target_level,
            )?;
        }

        if settle_oracle_warp {
            // Wait for the requested level to become live and unpaused. A mere
            // `Game_Mode != $0008` is not enough: the title-card Start press
            // can leave a stale pressed edge that pauses gameplay as soon as
            // pausing is enabled, and direct warp fallback must also prove it
            // reached the requested level instead of the pre-warp level.
            let mut stable_target_frames = 0usize;
            for step in 0..900 {
                capture_settle_trace_row(settle_trace.as_deref_mut(), emulator, "target_live", step);
                if target_level_is_live(emulator, target_level) {
                    stable_target_frames += 1;
                    if stable_target_frames >= 60 {
                        break;
                    }
                } else {
                    stable_target_frames = 0;
                }
                if read_game_mode(emulator) == 0x000C
                    && read_current_level_id(emulator) == u16::from(target_level)
                    && read_game_paused(emulator)
                {
                    tick_one_frame(emulator, renderer, audio, &start_input, save_writer)?;
                    clear_input_latches(emulator);
                    stable_target_frames = 0;
                }
                tick_one_frame(emulator, renderer, audio, &no_input, save_writer)?;
                clear_input_latches(emulator);
            }
        } else {
            // Legacy state-sweep bootstrap: preserve the established oracle
            // semantics used by the `1p-accurate` gate.
            for _ in 0..600 {
                if read_game_mode(emulator) != 0x0008 {
                    break;
                }
                tick_one_frame(emulator, renderer, audio, &no_input, save_writer)?;
            }
        }
    }

    Ok(())
}

fn should_force_target_level_for_probe(resolved: &ResolvedScenario) -> bool {
    resolved.start_position.is_some()
        || resolved.start_velocity.is_some()
        || resolved.camera_position.is_some()
        || resolved.playtest_form.is_some()
        || resolved.initial_diamond_count.is_some()
        || resolved.initial_lives.is_some()
}

fn drain_direct_probe_title_start_latch(
    emulator: &mut GenesisEmulator,
    renderer: &mut CaptureRenderer,
    audio: &mut AudioCollector,
    save_writer: &mut NullSaveWriter,
    target_level: u8,
) -> Result<()> {
    // ASM loc_74B0 pauses gameplay from a Start held edge after title-card
    // dismissal. Direct probe scenarios patch live WRAM after the bootstrap,
    // so consume one neutral rendered frame before scenario frame 0.
    let no_input = GenesisInputs::default();
    tick_one_frame(emulator, renderer, audio, &no_input, save_writer)?;
    clear_input_latches(emulator);
    force_warp_level_registers(emulator, target_level);
    Ok(())
}

fn settle_title_card_target_level(
    emulator: &mut GenesisEmulator,
    renderer: &mut CaptureRenderer,
    audio: &mut AudioCollector,
    save_writer: &mut NullSaveWriter,
    target_level: u8,
    mut settle_trace: Option<&mut Vec<OracleSettleTraceRow>>,
) -> Result<()> {
    let no_input = GenesisInputs::default();

    // Settled recording oracles must mirror native `enter_level_playtest`,
    // which starts from live gameplay. Booting to level 0 stops on the title
    // card. ASM loc_1AE2E waits on `bclr #Button_Start,(Ctrl_Pressed)`
    // before writing Game_Mode=$000C, so keep pulsing Start until that
    // object has consumed a real pressed edge.
    for _ in 0..120 {
        tick_one_frame(emulator, renderer, audio, &no_input, save_writer)?;
    }

    let start_input = GenesisInputs {
        p1: GenesisJoypadState {
            start: true,
            ..GenesisJoypadState::default()
        },
        p2: GenesisJoypadState::default(),
    };
    clear_input_latches(emulator);
    for _ in 0..120 {
        if read_game_mode(emulator) != 0x0008 {
            break;
        }
        tick_one_frame(emulator, renderer, audio, &start_input, save_writer)?;
    }
    clear_input_latches(emulator);

    let mut stable_target_frames = 0usize;
    for step in 0..900 {
        capture_settle_trace_row(settle_trace.as_deref_mut(), emulator, "title_target_live", step);
        if target_level_is_live(emulator, target_level) {
            stable_target_frames += 1;
            if stable_target_frames >= 60 {
                break;
            }
        } else {
            stable_target_frames = 0;
        }
        if read_game_mode(emulator) == 0x000C
            && read_current_level_id(emulator) == u16::from(target_level)
            && read_game_paused(emulator)
        {
            tick_one_frame(emulator, renderer, audio, &start_input, save_writer)?;
            clear_input_latches(emulator);
            stable_target_frames = 0;
        }
        tick_one_frame(emulator, renderer, audio, &no_input, save_writer)?;
        clear_input_latches(emulator);
    }

    Ok(())
}

fn force_warp_level_registers(emulator: &mut GenesisEmulator, target_level: u8) {
    write_emulator_wram_u16(emulator, CURRENT_LEVEL_ID_ADDR, u16::from(target_level));
    write_emulator_wram_u16(emulator, PLAYER_1_LEVEL_ID_ADDR, u16::from(target_level));
}

fn apply_probe_start_position(emulator: &mut GenesisEmulator, position: ScenarioStartPosition) {
    // ASM flag check path loc_BC4C -> loc_BC74 samples the live Kid object
    // x_pos/y_pos words. Probe scenarios patch only that live object position.
    // Match native `set_player_probe_position`: the direct live-state seed is
    // stationary, so clear ASM GfxObject offsets `x_vel=$26`/`y_vel=$2A`.
    write_emulator_wram_u32(emulator, KID_X_POS_ADDR, u32::from(position.x) << 16);
    write_emulator_wram_u32(emulator, KID_Y_POS_ADDR, u32::from(position.y) << 16);
    write_emulator_wram_u32(emulator, KID_X_VEL_ADDR, 0);
    write_emulator_wram_u32(emulator, KID_Y_VEL_ADDR, 0);
}

fn apply_probe_start_velocity(emulator: &mut GenesisEmulator, velocity: ScenarioStartVelocity) {
    // ASM GfxObject velocity offsets x_vel=$26/y_vel=$2A are signed 16.16
    // longwords. Probe scenarios may seed these after a direct position patch
    // to exercise upward collision windows without inventing a natural route.
    write_emulator_wram_u32(emulator, KID_X_VEL_ADDR, velocity.x as u32);
    write_emulator_wram_u32(emulator, KID_Y_VEL_ADDR, velocity.y as u32);
}

fn apply_probe_camera_position(emulator: &mut GenesisEmulator, position: ScenarioStartPosition) {
    // ASM variables.asm: Camera_X_pos/Camera_Y_pos are `$FFFFF81C`/`$FFFFF820`
    // long 16.16 words. Probe scenarios may patch them with Kid's live object
    // so mid-level visual evidence captures the same on-screen window.
    write_emulator_wram_u32(emulator, CAMERA_X_POS_ADDR, u32::from(position.x) << 16);
    write_emulator_wram_u32(emulator, CAMERA_Y_POS_ADDR, u32::from(position.y) << 16);
}

fn apply_probe_playtest_form(emulator: &mut GenesisEmulator, helmet_id: u16) -> Result<()> {
    // ASM variables.asm: Number_hitpoints/Current_Helmet_ID are `$FFFFFC40`
    // and `$FFFFFC46`. ASM sub_B41C (kid.asm:14331) also derives `$FFFFFA78`
    // collision half-width from Current_Helmet; seed it here because probe
    // playtest forms are direct live-state seeds, not helmet pickup transforms.
    // `Currently_transforming` is `$FFFFFA6D`; clear it for the same reason.
    let hitpoints = max_hitpoints_for_helmet_id(helmet_id)
        .ok_or_else(|| anyhow!("invalid playtest_form helmet id {helmet_id}"))?;
    let collision_half_width = collision_half_width_for_helmet_id(helmet_id)
        .ok_or_else(|| anyhow!("invalid playtest_form helmet id {helmet_id}"))?;
    write_emulator_wram_u16(emulator, CURRENT_HELMET_ID_ADDR, helmet_id);
    write_emulator_wram_u16(emulator, NUMBER_HITPOINTS_ADDR, hitpoints);
    write_emulator_wram_u16(
        emulator,
        PLAYER_COLLISION_HALF_WIDTH_ADDR,
        collision_half_width,
    );
    clear_probe_transform_residue(emulator);
    Ok(())
}

fn apply_probe_diamond_count(emulator: &mut GenesisEmulator, count: u16) {
    // ASM variables.asm: `Number_Diamonds` is `$FFFFFC42`.
    // `DiamondPower_Check` (kid.asm:44357) reads this word before spending
    // 20/50 diamonds, so diamond-power probes seed it directly after warp.
    write_emulator_wram_u16(emulator, NUMBER_DIAMONDS_ADDR, count);
}

fn apply_probe_lives_count(emulator: &mut GenesisEmulator, count: u16) {
    // ASM variables.asm: `Number_Lives` is `$FFFFFC3E`; player slot save
    // `Player_1_Lives` starts at `$FFFFFC54`. Keep both coherent for live
    // pause-menu probes that enter `Game_Paused_ChkStart` (kid.asm:6865).
    write_emulator_wram_u16(emulator, NUMBER_LIVES_ADDR, count);
    write_emulator_wram_u16(emulator, PLAYER_1_LIVES_ADDR, count);
}

fn clear_probe_transform_residue(emulator: &mut GenesisEmulator) {
    write_emulator_wram_u8(emulator, CURRENTLY_TRANSFORMING_ADDR, 0);
}

fn max_hitpoints_for_helmet_id(helmet_id: u16) -> Option<u16> {
    match helmet_id {
        0 => Some(2),
        6 => Some(5),
        1..=5 | 7..=9 => Some(3),
        _ => None,
    }
}

fn collision_half_width_for_helmet_id(helmet_id: u16) -> Option<u16> {
    // ASM unk_B408/sub_B41C (kid.asm:14307/14331): byte table indexed by
    // Current_Helmet * 2, then stored as a word in `$FFFFFA78`.
    match helmet_id {
        0 => Some(5),
        1 => Some(10),
        2..=4 | 6..=8 => Some(7),
        5 => Some(23),
        9 => Some(4),
        _ => None,
    }
}

fn clear_input_latches(emulator: &mut GenesisEmulator) {
    let mut mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
    mem.write(work_ram_index(CTRL_1_HELD_ADDR), 0);
    mem.write(work_ram_index(CTRL_1_PRESSED_ADDR), 0);
    mem.write(work_ram_index(CTRL_HELD_ADDR), 0);
    mem.write(work_ram_index(CTRL_PRESSED_ADDR), 0);
}

fn seed_input_latches(emulator: &mut GenesisEmulator, buttons: u8) {
    // ASM ReadJoypad loc_C4E publishes this frame's controller byte for the
    // next gameplay frame. Direct probe oracles capture before that publication,
    // so seed the frame-0 script byte after capture to keep frame 1 aligned.
    let mut mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
    mem.write(work_ram_index(CTRL_1_HELD_ADDR), buttons);
    mem.write(work_ram_index(CTRL_1_PRESSED_ADDR), buttons);
    mem.write(work_ram_index(CTRL_HELD_ADDR), buttons);
    mem.write(work_ram_index(CTRL_PRESSED_ADDR), buttons);
}

fn read_game_mode(emulator: &mut GenesisEmulator) -> u16 {
    let mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
    u16::from_be_bytes([
        mem.read(work_ram_index(GAME_MODE_ADDR)),
        mem.read(work_ram_index(GAME_MODE_ADDR) + 1),
    ])
}

fn read_current_level_id(emulator: &mut GenesisEmulator) -> u16 {
    let mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
    u16::from_be_bytes([
        mem.read(work_ram_index(CURRENT_LEVEL_ID_ADDR)),
        mem.read(work_ram_index(CURRENT_LEVEL_ID_ADDR) + 1),
    ])
}

fn read_game_paused(emulator: &mut GenesisEmulator) -> bool {
    let mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
    mem.read(work_ram_index(GAME_PAUSED_ADDR)) != 0
}

fn target_level_is_live(emulator: &mut GenesisEmulator, target_level: u8) -> bool {
    read_game_mode(emulator) == 0x000C
        && read_current_level_id(emulator) == u16::from(target_level)
        && !read_game_paused(emulator)
        // Settled recording cases compare from native `enter_level_playtest`,
        // where the live Kid object and object roster already exist. The ROM
        // can expose Game_Mode=$000C before Load_InGame has populated those
        // tables, so the oracle helper must not start scenario frame 0 on that
        // empty shell.
        && read_kid_x_pos(emulator) != 0
        && read_number_objects(emulator) != 0
}

fn capture_settle_trace_row(
    trace: Option<&mut Vec<OracleSettleTraceRow>>,
    emulator: &mut GenesisEmulator,
    phase: &'static str,
    step: usize,
) {
    let Some(trace) = trace else {
        return;
    };
    let work_ram = capture_memory(emulator, GenesisMemoryArea::WorkingRam, 0x1_0000);
    let capture_background = phase != "cold_boot";
    let object_slots_loc_e1e = {
        let rom = emulator.debug().memory_view(GenesisMemoryArea::CartridgeRom);
        object_slots_debug_loc_e1e(&work_ram, rom.as_ref())
    };
    let drips_object_slots_loc_33488 = drips_object_slots_debug_loc_33488(&work_ram);
    trace.push(OracleSettleTraceRow {
        phase,
        step,
        game_mode: read_emulator_wram_u16(emulator, GAME_MODE_ADDR),
        current_level_id: read_emulator_wram_u16(emulator, CURRENT_LEVEL_ID_ADDR),
        rng_offset: read_emulator_wram_u16(emulator, RNG_OFFSET_ADDR),
        rng_last_byte: rng_last_byte_from_emulator(emulator),
        // ASM Init_RNG/Get_RandomNumber_byte (kid.asm:1106-1204): retain the
        // full 55-byte ring because offset and last byte alone cannot prove
        // two collapsed hidden-shell boundaries have the same RNG state.
        rng_ram: work_ram[work_ram_index(RNG_RAM_START_ADDR)
            ..work_ram_index(RNG_RAM_END_ADDR)]
            .to_vec(),
        paused: read_emulator_wram_u8(emulator, GAME_PAUSED_ADDR) != 0,
        camera_x_pos: read_emulator_wram_u32(emulator, CAMERA_X_POS_ADDR),
        camera_y_pos: read_emulator_wram_u32(emulator, CAMERA_Y_POS_ADDR),
        kid_x_pos: read_emulator_wram_u32(emulator, KID_X_POS_ADDR),
        kid_y_pos: read_emulator_wram_u32(emulator, KID_Y_POS_ADDR),
        kid_x_vel: read_emulator_wram_i32(emulator, KID_X_VEL_ADDR),
        kid_y_vel: read_emulator_wram_i32(emulator, KID_Y_VEL_ADDR),
        number_objects: read_emulator_wram_u16(emulator, NUMBER_OBJECTS_ADDR),
        object_slots_loc_e1e,
        drips_object_slots_loc_33488,
        background_scroll_loc_3050a: capture_background
            .then(|| background_scroll_debug_loc_3050a(&work_ram))
            .unwrap_or_default(),
        background_row_offsets_loc_30744: capture_background
            .then(|| background_row_offsets_debug_loc_30744(&work_ram))
            .unwrap_or_default(),
        background_source_window_loc_30744: capture_background
            .then(|| background_source_window_debug_loc_30744(&work_ram))
            .unwrap_or_default(),
        background_source_absolute_loc_30744: capture_background
            .then(|| background_source_absolute_debug_loc_30744(&work_ram))
            .unwrap_or_default(),
        background_source_rows_loc_12848: capture_background
            .then(|| background_source_rows_debug_loc_12848(&work_ram))
            .unwrap_or_default(),
        background_plane_b_rows_loc_307d8: capture_background
            .then(|| background_plane_b_rows_debug_loc_307d8(emulator))
            .unwrap_or_default(),
        background_plane_b_spans_loc_307d8: capture_background

    });
}

fn read_kid_x_pos(emulator: &mut GenesisEmulator) -> u32 {
    let mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
    u32::from_be_bytes([
        mem.read(work_ram_index(KID_X_POS_ADDR)),
        mem.read(work_ram_index(KID_X_POS_ADDR) + 1),
        mem.read(work_ram_index(KID_X_POS_ADDR) + 2),
        mem.read(work_ram_index(KID_X_POS_ADDR) + 3),
    ])
}

fn read_number_objects(emulator: &mut GenesisEmulator) -> u16 {
    let mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
    u16::from_be_bytes([
        mem.read(work_ram_index(NUMBER_OBJECTS_ADDR)),
        mem.read(work_ram_index(NUMBER_OBJECTS_ADDR) + 1),
    ])
}

fn read_emulator_wram_u8(emulator: &mut GenesisEmulator, address: usize) -> u8 {
    let mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
    mem.read(work_ram_index(address))
}

fn rng_last_byte_from_emulator(emulator: &mut GenesisEmulator) -> u8 {
    let rng_offset = read_emulator_wram_u16(emulator, RNG_OFFSET_ADDR) as i16 as isize;
    read_emulator_wram_u8(emulator, (RNG_RAM_END_ADDR as isize + rng_offset) as usize)
}

fn read_emulator_wram_u16(emulator: &mut GenesisEmulator, address: usize) -> u16 {
    let mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
    u16::from_be_bytes([
        mem.read(work_ram_index(address)),
        mem.read(work_ram_index(address) + 1),
    ])
}

fn read_emulator_wram_u32(emulator: &mut GenesisEmulator, address: usize) -> u32 {
    let mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
    u32::from_be_bytes([
        mem.read(work_ram_index(address)),
        mem.read(work_ram_index(address) + 1),
        mem.read(work_ram_index(address) + 2),
        mem.read(work_ram_index(address) + 3),
    ])
}

fn read_emulator_wram_i32(emulator: &mut GenesisEmulator, address: usize) -> i32 {
    i32::from_be_bytes(read_emulator_wram_u32(emulator, address).to_be_bytes())
}

fn write_emulator_wram_u16(emulator: &mut GenesisEmulator, address: usize, value: u16) {
    let mut mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
    let [hi, lo] = value.to_be_bytes();
    mem.write(work_ram_index(address), hi);
    mem.write(work_ram_index(address) + 1, lo);
}

fn write_emulator_wram_u8(emulator: &mut GenesisEmulator, address: usize, value: u8) {
    let mut mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
    mem.write(work_ram_index(address), value);
}

fn write_emulator_wram_u32(emulator: &mut GenesisEmulator, address: usize, value: u32) {
    let mut mem = emulator.debug().memory_view(GenesisMemoryArea::WorkingRam);
    let [b0, b1, b2, b3] = value.to_be_bytes();
    mem.write(work_ram_index(address), b0);
    mem.write(work_ram_index(address) + 1, b1);
    mem.write(work_ram_index(address) + 2, b2);
    mem.write(work_ram_index(address) + 3, b3);
}


fn inputs_from_mask(mask: u8) -> GenesisInputs {
    let p1 = GenesisJoypadState {
        up: mask & (1 << 0) != 0,
        down: mask & (1 << 1) != 0,
        left: mask & (1 << 2) != 0,
        right: mask & (1 << 3) != 0,
        b: mask & (1 << 4) != 0,
        c: mask & (1 << 5) != 0,
        a: mask & (1 << 6) != 0,
        start: mask & (1 << 7) != 0,
        ..GenesisJoypadState::default()
    };
    GenesisInputs {
        p1,
        p2: GenesisJoypadState::default(),
    }
}

fn write_png(path: &Path, width: u32, height: u32, rgb: &[u8]) -> Result<()> {
    image::save_buffer_with_format(path, rgb, width, height, ColorType::Rgb8, ImageFormat::Png)
        .with_context(|| format!("failed to write {}", path.display()))
}

fn write_wav(path: &Path, samples: &[f32]) -> Result<()> {
    let spec = WavSpec {
        channels: 2,
        sample_rate: SAMPLE_RATE as u32,
        bits_per_sample: 32,
        sample_format: SampleFormat::Float,
    };
    let mut writer = WavWriter::create(path, spec)
        .with_context(|| format!("failed to create {}", path.display()))?;
    for sample in samples {
        writer
            .write_sample(*sample)
            .with_context(|| format!("failed to write samples to {}", path.display()))?;
    }
    writer
        .finalize()
        .with_context(|| format!("failed to finalise {}", path.display()))
}

fn sha1_hex(bytes: &[u8]) -> String {
    let mut hasher = Sha1::new();
    hasher.update(bytes);
    let digest = hasher.finalize();
    let mut hex = String::with_capacity(digest.len() * 2);
    for byte in digest {
        hex.push_str(&format!("{byte:02x}"));
    }
    hex
}
